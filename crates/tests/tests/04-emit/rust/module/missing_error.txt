/// Typed publication of a region-owned result graph from walk steps.
mod publish {
    use super::*;

    /// The logical result schema shared with the runtime walk and JSON.
    pub(super) const SCHEMA: rt::Schema<'static> = rt::Schema {
        types: &[
            rt::Ty::Record { fields: rt::Span::new(0, 1) },
            rt::Ty::Record { fields: rt::Span::new(1, 1) },
            rt::Ty::Node,
        ],
        fields: &[
            rt::Field { name: "err", ty: 2 },
            rt::Field { name: "gap", ty: 2 },
        ],
        cases: &[],
        members: &[
            rt::Member::new(0, 0),
            rt::Member::new(1, 0),
        ],
    };

    /// Native layout of each schema type, for sizing result blocks.
    pub(super) const LAYOUTS: &[::std::alloc::Layout] = &[
        ::std::alloc::Layout::new::<Err<'static>>(),
        ::std::alloc::Layout::new::<Q<'static>>(),
        ::std::alloc::Layout::new::<Node<'static>>(),
    ];

    /// One value stack per schema type, plus one writer stack per list type.
    pub(super) struct Stacks0<'a, 't, 's> {
        pub(super) t0: rt::Buffer<Err<'t>>,
        pub(super) t2: rt::Buffer<Node<'t>>,
        _lifetimes: ::core::marker::PhantomData<(&'a (), &'t (), &'s ())>,
    }

    impl<'a, 't, 's> Stacks0<'a, 't, 's> {
        pub(super) fn new() -> Self {
            Self {
                t0: rt::Buffer::new(),
                t2: rt::Buffer::new(),
                _lifetimes: ::core::marker::PhantomData,
            }
        }

        /// Apply walk steps until the walk ends.
        pub(super) fn run(
            &mut self,
            walk: &mut rt::Walk<'_, '_, 't, 's>,
            _scope: &rt::Scope<'a>,
            meter: &mut rt::Meter<'_>,
        ) -> Result<(), rt::LimitExceeded> {
            while let Some(step) = walk.next(meter)? {
                match step {
                    rt::Step::Close { ty: 0, .. } => self.close_0(meter)?,
                    rt::Step::Node { ty: 2, node } => self.t2.push(meter, node)?,
                    rt::Step::Open { .. } | rt::Step::Field { .. } => {}
                    other => unreachable!("the schema admits no step {other:?}"),
                }
            }
            Ok(())
        }

        #[inline(never)]
        fn close_0(&mut self, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let err = pop(&mut self.t2);
            self.t0.push(meter, Err { err })
        }
    }


    /// One value stack per schema type, plus one writer stack per list type.
    pub(super) struct Stacks1<'a, 't, 's> {
        pub(super) t1: rt::Buffer<Q<'t>>,
        pub(super) t2: rt::Buffer<Node<'t>>,
        _lifetimes: ::core::marker::PhantomData<(&'a (), &'t (), &'s ())>,
    }

    impl<'a, 't, 's> Stacks1<'a, 't, 's> {
        pub(super) fn new() -> Self {
            Self {
                t1: rt::Buffer::new(),
                t2: rt::Buffer::new(),
                _lifetimes: ::core::marker::PhantomData,
            }
        }

        /// Apply walk steps until the walk ends.
        pub(super) fn run(
            &mut self,
            walk: &mut rt::Walk<'_, '_, 't, 's>,
            _scope: &rt::Scope<'a>,
            meter: &mut rt::Meter<'_>,
        ) -> Result<(), rt::LimitExceeded> {
            while let Some(step) = walk.next(meter)? {
                match step {
                    rt::Step::Close { ty: 1, .. } => self.close_1(meter)?,
                    rt::Step::Node { ty: 2, node } => self.t2.push(meter, node)?,
                    rt::Step::Open { .. } | rt::Step::Field { .. } => {}
                    other => unreachable!("the schema admits no step {other:?}"),
                }
            }
            Ok(())
        }

        #[inline(never)]
        fn close_1(&mut self, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let gap = pop(&mut self.t2);
            self.t1.push(meter, Q { gap })
        }
    }

    pub(super) fn pop<T>(stack: &mut rt::Buffer<T>) -> T {
        stack.pop().expect("the walk produced every value before its parent")
    }
}

/// The compiled matcher: engine machinery shielded from the query's
/// type namespace.
mod matcher {
    use ::plotnik_runtime as rt;

    /// The limit policy compiled into the safe entrypoints, resolved against
    /// each input's node count. Chosen at generation time, never at the call
    /// site: the query is trusted, the input is not.
    const LIMITS: rt::RuntimeLimitSpec = rt::RuntimeLimitSpec {
        fuel_limit: rt::Limit::Auto,
        memory: rt::Limit::Auto,
    };

    /// Node-kind ids baked into the candidate checks: `(id, name, is_named)`
    /// as the generation-time grammar defines them.
    const EXPECTED_KINDS: &[(u16, &str, bool)] = &[
        (9, "}", false),
        (134, "program", true),
        (155, "statement_block", true),
        (157, "if_statement", true),
    ];

    /// Field ids baked into the field checks: `(id, name)`.
    const EXPECTED_FIELDS: &[(u16, &str)] = &[(9, "consequence")];

    /// A parser built from any other grammar version could renumber the baked
    /// kind/field ids and silently mis-match, so mismatches panic: version skew
    /// between the generation-time grammar and the runtime parser is a build
    /// mistake, not a runtime condition to recover from.
    pub(super) fn verify_language(node: rt::Node<'_>) {
        let language = node.language();
        for &(id, name, named) in EXPECTED_KINDS {
            let found = language.node_kind_for_id(id);
            if found != Some(name) || language.node_kind_is_named(id) != named {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where node kind {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
        for &(id, name) in EXPECTED_FIELDS {
            let found = language.field_name_for_id(id);
            if found != Some(name) {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where field {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
    }

    const F_CONSEQUENCE: rt::NodeFieldId = rt::NodeFieldId::from_raw(9);

    // Dense runtime state ids, in NFA label order.
    // Err:
    const S01_ERR: u16 = 0;
    const S02_ERR: u16 = 1;
    const S03_ERR: u16 = 2;
    const S04_ERR: u16 = 3;
    // Q:
    const S06_Q: u16 = 4;
    const S07_Q: u16 = 5;
    const S09_Q: u16 = 6;
    const S11_Q: u16 = 7;
    const S12_Q: u16 = 8;
    const S13_Q: u16 = 9;

    pub(super) struct ErrEntry;

    impl rt::Entry for ErrEntry {
        const LIMITS: rt::RuntimeLimitSpec = LIMITS;

        fn verify_language(node: rt::Node<'_>) {
            verify_language(node);
        }

        fn run<'t>(
            eng: &mut rt::Engine<'t>,
            m: &mut rt::Meter<'_>,
            source: &str,
        ) -> Result<bool, rt::LimitExceeded> {
            run(eng, m, source, S02_ERR, EntryBoundary::new(true, false))
        }
    }

    pub(super) struct QEntry;

    impl rt::Entry for QEntry {
        const LIMITS: rt::RuntimeLimitSpec = LIMITS;

        fn verify_language(node: rt::Node<'_>) {
            verify_language(node);
        }

        fn run<'t>(
            eng: &mut rt::Engine<'t>,
            m: &mut rt::Meter<'_>,
            source: &str,
        ) -> Result<bool, rt::LimitExceeded> {
            run(eng, m, source, S07_Q, EntryBoundary::new(true, false))
        }
    }
    #[derive(Clone, Copy)]
    struct EntryBoundary {
        record: bool,
        node: bool,
    }

    impl EntryBoundary {
        const fn new(record: bool, node: bool) -> Self {
            Self { record, node }
        }

        fn begin(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordOpen)?;
            }
            Ok(())
        }

        fn finish(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.node {
                eng.emit(m, |cursor| rt::JournalEvent::Node(cursor.node()))?;
                return Ok(());
            }
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordClose)?;
            }
            Ok(())
        }
    }

    /// What a dispatched state hands back to the driver loop.
    enum Flow {
        /// Continue at this state.
        Jump(u16),
        /// The entrypoint accepted; the match journal is committed.
        Accept,
        /// The state failed; unwind the checkpoint stack.
        Backtrack,
    }

    /// How the backtrack unwind resumed execution.
    enum Unwound {
        Resumed(u16),
        Accepted,
        NoMatch,
    }

    fn run<'t>(
        eng: &mut rt::Engine<'t>,
        m: &mut rt::Meter<'_>,
        source: &str,
        entry: u16,
        boundary: EntryBoundary,
    ) -> Result<bool, rt::LimitExceeded> {
        boundary.begin(eng, m)?;
        let mut ip = entry;
        loop {
            m.charge(1)?;
            match dispatch(eng, m, source, ip)? {
                Flow::Jump(next) => ip = next,
                Flow::Accept => {
                    boundary.finish(eng, m)?;
                    eng.accepted_tape();
                    return Ok(true);
                }
                Flow::Backtrack => match backtrack(eng, m, source)? {
                    Unwound::Resumed(next) => ip = next,
                    Unwound::Accepted => {
                        boundary.finish(eng, m)?;
                        eng.accepted_tape();
                        return Ok(true);
                    }
                    Unwound::NoMatch => return Ok(false),
                },
            }
        }
    }

    fn dispatch<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Flow, rt::LimitExceeded> {
        Ok(match ip {
            //   01                                        ▶
            S01_ERR => Flow::Accept,
            //   02   !   (program)                        04
            S02_ERR => 'state: {
                if !cand_s02_err(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S04_ERR)
            }
            //   03  ─‣┘  _                                01
            S03_ERR => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Up(1))?.is_none() {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S01_ERR)
            }
            //   04  └‣─  (ERROR) [RecordSet(err) Node]    03
            S04_ERR => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                loop {
                    if cand_s04_err(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        break 'state Flow::Backtrack;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S04_ERR,
                    ))?;
                }
                finish_s04_err(eng, m)?
            }
            //   06                                        ▶
            S06_Q => Flow::Accept,
            //   07   !   (program)                        09
            S07_Q => 'state: {
                if !cand_s07_q(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S09_Q)
            }
            //   09  └‣─  (if_statement)                   11
            S09_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                loop {
                    if cand_s09_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        break 'state Flow::Backtrack;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S09_Q,
                    ))?;
                }
                finish_s09_q(eng, m)?
            }
            //   11  └‣─  consequence: (statement_block)   13
            S11_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                loop {
                    if cand_s11_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        break 'state Flow::Backtrack;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S11_Q,
                    ))?;
                }
                finish_s11_q(eng, m)?
            }
            //   12  ─‣┘³ _                                06
            S12_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Up(3))?.is_none() {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S06_Q)
            }
            //   13  └‣─  (MISSING "}") [RecordSet(gap) Node]  12
            S13_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                loop {
                    if cand_s13_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        break 'state Flow::Backtrack;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S13_Q,
                    ))?;
                }
                finish_s13_q(eng, m)?
            }
            _ => unreachable!("ip {ip} is not a generated state"),
        })
    }

    /// `S02_ERR` candidate: `(program)`.
    #[inline]
    fn cand_s02_err(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (program)
        if node.kind_id() != 134 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S04_ERR` candidate: `(ERROR)`.
    #[inline]
    fn cand_s04_err(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (ERROR)
        if node.kind_id() != 65535 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S07_Q` candidate: `(program)`.
    #[inline]
    fn cand_s07_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (program)
        if node.kind_id() != 134 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S09_Q` candidate: `(if_statement)`.
    #[inline]
    fn cand_s09_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (if_statement)
        if node.kind_id() != 157 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S11_Q` candidate: `consequence: (statement_block)`.
    #[inline]
    fn cand_s11_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (statement_block)
        if node.kind_id() != 155 || !node.is_named() {
            return Ok(false);
        }
        // consequence:
        if !eng.cursor_mut().has_field(m, F_CONSEQUENCE)? {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S13_Q` candidate: `(MISSING "}")`.
    #[inline]
    fn cand_s13_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // "}"
        if node.kind_id() != 9 || node.is_named() {
            return Ok(false);
        }
        // (MISSING …): only nodes the parser inserted during error recovery
        if !node.is_missing() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S04_ERR` post-acceptance: effects, then successor dispatch. Shared by the dispatch
    /// path and the match-retry resume, so a retried candidate emits exactly
    /// what the original acceptance would have emitted.
    #[inline]
    fn finish_s04_err(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<Flow, rt::LimitExceeded> {
        let flow = {
            eng.emit(m, |_| rt::JournalEvent::RecordSet(0))?; // RecordSet(err)
            eng.emit(m, |c| rt::JournalEvent::Node(c.node()))?;
            Flow::Jump(S03_ERR)
        };
        Ok(flow)
    }

    /// `S09_Q` post-acceptance: effects, then successor dispatch. Shared by the dispatch
    /// path and the match-retry resume, so a retried candidate emits exactly
    /// what the original acceptance would have emitted.
    #[inline]
    fn finish_s09_q(_eng: &mut rt::Engine<'_>, _m: &mut rt::Meter<'_>) -> Result<Flow, rt::LimitExceeded> {
        let flow = {
            Flow::Jump(S11_Q)
        };
        Ok(flow)
    }

    /// `S11_Q` post-acceptance: effects, then successor dispatch. Shared by the dispatch
    /// path and the match-retry resume, so a retried candidate emits exactly
    /// what the original acceptance would have emitted.
    #[inline]
    fn finish_s11_q(_eng: &mut rt::Engine<'_>, _m: &mut rt::Meter<'_>) -> Result<Flow, rt::LimitExceeded> {
        let flow = {
            Flow::Jump(S13_Q)
        };
        Ok(flow)
    }

    /// `S13_Q` post-acceptance: effects, then successor dispatch. Shared by the dispatch
    /// path and the match-retry resume, so a retried candidate emits exactly
    /// what the original acceptance would have emitted.
    #[inline]
    fn finish_s13_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<Flow, rt::LimitExceeded> {
        let flow = {
            eng.emit(m, |_| rt::JournalEvent::RecordSet(1))?; // RecordSet(gap)
            eng.emit(m, |c| rt::JournalEvent::Node(c.node()))?;
            Flow::Jump(S12_Q)
        };
        Ok(flow)
    }

    /// Unwind the checkpoint stack: successor checkpoints resume dispatch, Call and
    /// Match checkpoints advance their sibling search and re-enter. Loops, never
    /// recurses — a run of exhausted retries unwinds in one call.
    fn backtrack<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str) -> Result<Unwound, rt::LimitExceeded> {
        'unwind: loop {
            m.charge(1)?;
            let Some((cp, snapshot)) = eng.pop_checkpoint() else {
                return Ok(Unwound::NoMatch);
            };
            eng.restore_checkpoint_state(m, cp.state, snapshot)?;

            match cp.resume {
                rt::Resume::Successor => return Ok(Unwound::Resumed(cp.ip)),

                rt::Resume::Call(_) => unreachable!("matcher has no call checkpoints"),

                // Match retry: advance past the accepted-but-failed candidate and
                // re-run that state's sibling search from there.
                rt::Resume::Match => match match_retry(eng, m, _source, cp.ip)? {
                    Some(Flow::Jump(next)) => return Ok(Unwound::Resumed(next)),
                    Some(Flow::Accept) => return Ok(Unwound::Accepted),
                    Some(Flow::Backtrack) => unreachable!("finish never backtracks"),
                    None => continue 'unwind,
                },
            }
        }
    }

    fn match_retry<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Option<Flow>, rt::LimitExceeded> {
        match ip {
            S04_ERR => {
                if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                    return Ok(None);
                }
                loop {
                    if cand_s04_err(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        return Ok(None);
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S04_ERR,
                    ))?;
                }
                finish_s04_err(eng, m).map(Some)
            }
            S09_Q => {
                if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                    return Ok(None);
                }
                loop {
                    if cand_s09_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        return Ok(None);
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S09_Q,
                    ))?;
                }
                finish_s09_q(eng, m).map(Some)
            }
            S11_Q => {
                if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                    return Ok(None);
                }
                loop {
                    if cand_s11_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        return Ok(None);
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S11_Q,
                    ))?;
                }
                finish_s11_q(eng, m).map(Some)
            }
            S13_Q => {
                if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                    return Ok(None);
                }
                loop {
                    if cand_s13_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        return Ok(None);
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S13_Q,
                    ))?;
                }
                finish_s13_q(eng, m).map(Some)
            }
            _ => unreachable!("match-retry checkpoint ip {ip} must address a sibling-search Match"),
        }
    }
}

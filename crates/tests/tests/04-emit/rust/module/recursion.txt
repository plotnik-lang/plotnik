// Generated Plotnik query module: typed result types, supplied-node execution
// points, per-type result decoders, and the compiled matcher (`mod matcher`).
// Matcher states mirror the NFA dump's labels 1:1 (`S{label}_{DEF}`), and every
// dispatch arm carries its instruction in the dump format.

use ::plotnik_runtime as rt;

pub const REQUIRED_RUNTIME_ABI: u32 = 0;
const _: () = assert!(
    rt::RUNTIME_ABI == REQUIRED_RUNTIME_ABI,
    "Plotnik runtime ABI mismatch"
);

use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub enum Expr<'r, 't> {
    Paren { inner: &'r Expr<'r, 't> },
    Id { id: Node<'t> },
}

#[derive(Clone, Copy)]
pub struct Q<'r, 't> {
    pub expr: Expr<'r, 't>,
}

const _: () = assert!(::core::mem::size_of::<Expr<'static, 'static>>() <= rt::GENERATED_VALUE_CAP);
const _: () = assert!(::core::mem::size_of::<Q<'static, 'static>>() <= rt::GENERATED_VALUE_CAP);

rt::result_family!(Expr<'r, 't>);
rt::result_family!(Q<'r, 't>);

impl rt::json::Json for Expr<'_, '_> {
    fn shape(&self) -> rt::json::Shape<'_> {
        match self {
            Expr::Paren { .. } => rt::json::Shape::Variant { tag: "Paren", data: Some(self) },
            Expr::Id { .. } => rt::json::Shape::Variant { tag: "Id", data: Some(self) },
        }
    }
}

impl rt::json::Fields for Expr<'_, '_> {
    fn count(&self) -> usize {
        match self {
            Expr::Paren { .. } => 1,
            Expr::Id { .. } => 1,
        }
    }

    fn field(&self, index: usize) -> (&'static str, &dyn rt::json::Json) {
        match (self, index) {
            (Expr::Paren { inner, .. }, 0) => ("inner", inner),
            (Expr::Id { id, .. }, 0) => ("id", id),
            _ => unreachable!("`Expr` case has no field {index}"),
        }
    }
}

impl rt::json::Json for Q<'_, '_> {
    fn shape(&self) -> rt::json::Shape<'_> {
        rt::json::Shape::Record(self)
    }
}

impl rt::json::Fields for Q<'_, '_> {
    fn count(&self) -> usize {
        1
    }

    fn field(&self, index: usize) -> (&'static str, &dyn rt::json::Json) {
        match index {
            0 => ("expr", &self.expr),
            _ => unreachable!("`Q` has 1 fields"),
        }
    }
}

impl<'r, 't> Expr<'r, 't> {
    /// Apply `Expr` once at a suitable node. A wrong node kind is no-match.
    pub fn exec<'s>(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<rt::Match<Self>>, rt::LimitExceeded> {
        rt::run::exec::<Self, matcher::ExprEntry>(node, source)
    }

    /// Search inclusively in source-order DFS preorder, skipping descendants of matches.
    /// Each item owns its result. Matching and traversal share compiled-in limits.
    /// An error ends iteration.
    pub fn each<'s>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<
        Item = ::core::result::Result<rt::Match<Self>, rt::LimitExceeded>,
    > + use<'r, 't, 's> {
        rt::run::each::<Self, matcher::ExprEntry>(scope, source)
    }

    /// Consume the first item of `each`.
    pub fn find<'s>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<rt::Match<Self>>, rt::LimitExceeded> {
        Self::each(scope, source).next().transpose()
    }

    /// Collect `each`, returning an error rather than a partial collection.
    pub fn find_all<'s>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<rt::Matches<Self>, rt::LimitExceeded> {
        rt::run::find_all::<Self, matcher::ExprEntry>(scope, source)
    }
}

/// Typed construction of result values from walk steps.
mod publish {
    use super::*;

    /// The logical result schema shared with the runtime walk and JSON.
    pub(super) const SCHEMA: rt::Schema<'static> = rt::Schema {
        types: &[
            rt::Ty::Variant { cases: rt::Span::new(0, 2) },
            rt::Ty::Record { fields: rt::Span::new(0, 1) },
            rt::Ty::Record { fields: rt::Span::new(1, 1) },
            rt::Ty::Ref(0),
            rt::Ty::Record { fields: rt::Span::new(2, 1) },
            rt::Ty::Node,
        ],
        fields: &[
            rt::Field { name: "expr", ty: 0 },
            rt::Field { name: "inner", ty: 3 },
            rt::Field { name: "id", ty: 5 },
        ],
        cases: &[
            rt::Case { name: "Paren", payload: 2 },
            rt::Case { name: "Id", payload: 4 },
        ],
        members: &[
            rt::Member::new(2, 0),
            rt::Member::new(4, 0),
            rt::Member::new(0, 0),
            rt::Member::new(0, 1),
            rt::Member::new(1, 0),
        ],
    };

    /// Native layout of each schema type, for sizing result blocks.
    pub(super) const LAYOUTS: &[::std::alloc::Layout] = &[
        ::std::alloc::Layout::new::<Expr<'static, 'static>>(),
        ::std::alloc::Layout::new::<Q<'static, 'static>>(),
        ::std::alloc::Layout::new::<()>(),
        ::std::alloc::Layout::new::<&'static Expr<'static, 'static>>(),
        ::std::alloc::Layout::new::<()>(),
        ::std::alloc::Layout::new::<Node<'static>>(),
    ];

    /// One value stack per schema type, plus one writer stack per list type.
    pub(super) struct Stacks0<'a, 't, 's> {
        pub(super) t0: rt::Buffer<Expr<'a, 't>>,
        pub(super) t3: rt::Buffer<&'a Expr<'a, 't>>,
        pub(super) t5: rt::Buffer<Node<'t>>,
        _lifetimes: ::core::marker::PhantomData<(&'a (), &'t (), &'s ())>,
    }

    impl<'a, 't, 's> Stacks0<'a, 't, 's> {
        pub(super) fn new() -> Self {
            Self {
                t0: rt::Buffer::new(),
                t3: rt::Buffer::new(),
                t5: rt::Buffer::new(),
                _lifetimes: ::core::marker::PhantomData,
            }
        }

        /// Apply walk steps until the walk ends.
        pub(super) fn run(
            &mut self,
            walk: &mut rt::Walk<'_, '_, 't, 's>,
            _scope: &rt::Scope<'a>,
            meter: &mut rt::Meter<'_>,
        ) -> Result<(), rt::LimitExceeded> {
            while let Some(step) = walk.next(meter)? {
                match step {
                    rt::Step::Close { ty: 0, case } => self.close_0(case, meter)?,
                    rt::Step::Close { ty: 3, .. } => self.close_3(_scope, meter)?,
                    rt::Step::Node { ty: 5, node } => self.t5.push(meter, node)?,
                    rt::Step::Open { .. } | rt::Step::Field { .. } => {}
                    other => unreachable!("the schema admits no step {other:?}"),
                }
            }
            Ok(())
        }

        #[inline(never)]
        fn close_0(&mut self, case: u32, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let value = match case {
                0 => {
                    let inner = pop(&mut self.t3);
                    Expr::Paren { inner }
                }
                1 => {
                    let id = pop(&mut self.t5);
                    Expr::Id { id }
                }
                _ => unreachable!("`Expr` has no case {case}"),
            };
            self.t0.push(meter, value)
        }

        #[inline(never)]
        fn close_3(&mut self, scope: &rt::Scope<'a>, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let value = pop(&mut self.t0);
            let value = scope.push(meter, value)?;
            self.t3.push(meter, value)
        }
    }


    /// One value stack per schema type, plus one writer stack per list type.
    pub(super) struct Stacks1<'a, 't, 's> {
        pub(super) t0: rt::Buffer<Expr<'a, 't>>,
        pub(super) t1: rt::Buffer<Q<'a, 't>>,
        pub(super) t3: rt::Buffer<&'a Expr<'a, 't>>,
        pub(super) t5: rt::Buffer<Node<'t>>,
        _lifetimes: ::core::marker::PhantomData<(&'a (), &'t (), &'s ())>,
    }

    impl<'a, 't, 's> Stacks1<'a, 't, 's> {
        pub(super) fn new() -> Self {
            Self {
                t0: rt::Buffer::new(),
                t1: rt::Buffer::new(),
                t3: rt::Buffer::new(),
                t5: rt::Buffer::new(),
                _lifetimes: ::core::marker::PhantomData,
            }
        }

        /// Apply walk steps until the walk ends.
        pub(super) fn run(
            &mut self,
            walk: &mut rt::Walk<'_, '_, 't, 's>,
            _scope: &rt::Scope<'a>,
            meter: &mut rt::Meter<'_>,
        ) -> Result<(), rt::LimitExceeded> {
            while let Some(step) = walk.next(meter)? {
                match step {
                    rt::Step::Close { ty: 0, case } => self.close_0(case, meter)?,
                    rt::Step::Close { ty: 1, .. } => self.close_1(meter)?,
                    rt::Step::Close { ty: 3, .. } => self.close_3(_scope, meter)?,
                    rt::Step::Node { ty: 5, node } => self.t5.push(meter, node)?,
                    rt::Step::Open { .. } | rt::Step::Field { .. } => {}
                    other => unreachable!("the schema admits no step {other:?}"),
                }
            }
            Ok(())
        }

        #[inline(never)]
        fn close_0(&mut self, case: u32, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let value = match case {
                0 => {
                    let inner = pop(&mut self.t3);
                    Expr::Paren { inner }
                }
                1 => {
                    let id = pop(&mut self.t5);
                    Expr::Id { id }
                }
                _ => unreachable!("`Expr` has no case {case}"),
            };
            self.t0.push(meter, value)
        }

        #[inline(never)]
        fn close_1(&mut self, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let expr = pop(&mut self.t0);
            self.t1.push(meter, Q { expr })
        }

        #[inline(never)]
        fn close_3(&mut self, scope: &rt::Scope<'a>, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let value = pop(&mut self.t0);
            let value = scope.push(meter, value)?;
            self.t3.push(meter, value)
        }
    }

    pub(super) fn pop<T>(stack: &mut rt::Buffer<T>) -> T {
        stack.pop().expect("the walk produced every value before its parent")
    }
}

/// The compiled matcher: engine machinery shielded from the query's
/// type namespace.
mod matcher {
    use ::plotnik_runtime as rt;

    /// The limit policy compiled into the safe entrypoints, resolved against
    /// each input's node count. Chosen at generation time, never at the call
    /// site: the query is trusted, the input is not.
    const LIMITS: rt::RuntimeLimitSpec = rt::RuntimeLimitSpec {
        fuel_limit: rt::Limit::Auto,
        memory: rt::Limit::Auto,
    };

    /// Node-kind ids baked into the candidate checks: `(id, name, is_named)`
    /// as the generation-time grammar defines them.
    const EXPECTED_KINDS: &[(u16, &str, bool)] = &[
        (1, "identifier", true),
        (134, "program", true),
        (150, "expression_statement", true),
        (178, "parenthesized_expression", true),
    ];

    /// Field ids baked into the field checks: `(id, name)`.
    const EXPECTED_FIELDS: &[(u16, &str)] = &[];

    /// A parser built from any other grammar version could renumber the baked
    /// kind/field ids and silently mis-match, so mismatches panic: version skew
    /// between the generation-time grammar and the runtime parser is a build
    /// mistake, not a runtime condition to recover from.
    pub(super) fn verify_language(node: rt::Node<'_>) {
        let language = node.language();
        for &(id, name, named) in EXPECTED_KINDS {
            let found = language.node_kind_for_id(id);
            if found != Some(name) || language.node_kind_is_named(id) != named {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where node kind {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
        for &(id, name) in EXPECTED_FIELDS {
            let found = language.field_name_for_id(id);
            if found != Some(name) {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where field {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
    }

    // Dense runtime state ids, in NFA label order.
    // Expr:
    const S00_EXPR: u16 = 0;
    const S01_EXPR: u16 = 1;
    const S02_EXPR: u16 = 2;
    const S04_EXPR: u16 = 3;
    const S06_EXPR: u16 = 4;
    const S08_EXPR: u16 = 5;
    // Q:
    const S11_Q: u16 = 6;
    const S12_Q: u16 = 7;
    const S14_Q: u16 = 8;
    const S15_Q: u16 = 9;
    const S17_Q: u16 = 10;

    /// Resolve a callee-local return port through the immutable call-site map.
    #[inline]
    fn call_return(call_site: u16, port: rt::PortId) -> u16 {
        match (call_site, port.to_byte()) {
            (S06_EXPR, 0) => S04_EXPR,
            (S17_Q, 0) => S15_Q,
            _ => unreachable!(
                "call site {call_site} has no return port {}",
                port.to_byte()
            ),
        }
    }

    pub(super) struct ExprEntry;

    impl rt::Entry for ExprEntry {
        const LIMITS: rt::RuntimeLimitSpec = LIMITS;

        fn verify_language(node: rt::Node<'_>) {
            verify_language(node);
        }

        fn run<'t>(
            eng: &mut rt::Engine<'t>,
            m: &mut rt::Meter<'_>,
            source: &str,
        ) -> Result<bool, rt::LimitExceeded> {
            run(eng, m, source, S00_EXPR, EntryBoundary::new(false, false))
        }
    }

    pub(super) struct QEntry;

    impl rt::Entry for QEntry {
        const LIMITS: rt::RuntimeLimitSpec = LIMITS;

        fn verify_language(node: rt::Node<'_>) {
            verify_language(node);
        }

        fn run<'t>(
            eng: &mut rt::Engine<'t>,
            m: &mut rt::Meter<'_>,
            source: &str,
        ) -> Result<bool, rt::LimitExceeded> {
            run(eng, m, source, S12_Q, EntryBoundary::new(true, false))
        }
    }
    #[derive(Clone, Copy)]
    struct EntryBoundary {
        record: bool,
        node: bool,
    }

    impl EntryBoundary {
        const fn new(record: bool, node: bool) -> Self {
            Self { record, node }
        }

        fn begin(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordOpen)?;
            }
            Ok(())
        }

        fn finish(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.node {
                eng.emit(m, |cursor| rt::JournalEvent::Node(cursor.node()))?;
                return Ok(());
            }
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordClose)?;
            }
            Ok(())
        }
    }

    /// What a dispatched state hands back to the driver loop.
    enum Flow {
        /// Continue at this state.
        Jump(u16),
        /// The entrypoint accepted; the match journal is committed.
        Accept,
        /// The state failed; unwind the checkpoint stack.
        Backtrack,
    }

    /// How the backtrack unwind resumed execution.
    enum Unwound {
        Resumed(u16),
        Accepted,
        NoMatch,
    }

    fn run<'t>(
        eng: &mut rt::Engine<'t>,
        m: &mut rt::Meter<'_>,
        source: &str,
        entry: u16,
        boundary: EntryBoundary,
    ) -> Result<bool, rt::LimitExceeded> {
        boundary.begin(eng, m)?;
        let mut ip = entry;
        loop {
            m.charge(1)?;
            match dispatch(eng, m, source, ip)? {
                Flow::Jump(next) => ip = next,
                Flow::Accept => {
                    boundary.finish(eng, m)?;
                    eng.accepted_tape();
                    return Ok(true);
                }
                Flow::Backtrack => match backtrack(eng, m, source)? {
                    Unwound::Resumed(next) => ip = next,
                    Unwound::Accepted => {
                        boundary.finish(eng, m)?;
                        eng.accepted_tape();
                        return Ok(true);
                    }
                    Unwound::NoMatch => return Ok(false),
                },
            }
        }
    }

    fn dispatch<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Flow, rt::LimitExceeded> {
        Ok(match ip {
            //   00  -ε-                                   02, 08
            S00_EXPR => {
                eng.push_successors(m, &[S08_EXPR])?;
                Flow::Jump(S02_EXPR)
            }
            //   01                                        ▶
            S01_EXPR => {
                if eng.frames_empty() {
                    Flow::Accept
                } else {
                    let call_site = eng.exit_frame();
                    Flow::Jump(call_return(call_site, rt::PortId::from_raw(0)))
                }
            }
            //   02   !   (parenthesized_expression) [VariantOpen(Paren) RecordSet(inner)]  06
            S02_EXPR => 'state: {
                if !cand_s02_expr(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                eng.emit(m, |_| rt::JournalEvent::VariantOpen(2))?; // VariantOpen(Paren)
                eng.emit(m, |_| rt::JournalEvent::RecordSet(0))?; // RecordSet(inner)
                Flow::Jump(S06_EXPR)
            }
            //   04  ─‣┘  _ [VariantClose]                 01
            S04_EXPR => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Up(1))?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.emit(m, |_| rt::JournalEvent::VariantClose)?;
                Flow::Jump(S01_EXPR)
            }
            //   06  └‣─  (Expr)                           00 : 04
            S06_EXPR => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_checkpoint(m, rt::Checkpoint::call_retry(
                    eng.checkpoint_state(),
                    S06_EXPR,
                    rt::CallResume {
                        target: S00_EXPR,
                        call_site: S06_EXPR,
                        field: None,
                        policy: rt::SkipPolicy::Any,
                    },
                ))?;
                eng.enter_frame(m, S06_EXPR)?;
                Flow::Jump(S00_EXPR)
            }
            //   08   !   (identifier) [VariantOpen(Id) RecordSet(id) Node VariantClose]  01
            S08_EXPR => 'state: {
                if !cand_s08_expr(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                eng.emit(m, |_| rt::JournalEvent::VariantOpen(3))?; // VariantOpen(Id)
                eng.emit(m, |_| rt::JournalEvent::RecordSet(1))?; // RecordSet(id)
                eng.emit(m, |c| rt::JournalEvent::Node(c.node()))?;
                eng.emit(m, |_| rt::JournalEvent::VariantClose)?;
                Flow::Jump(S01_EXPR)
            }
            //   11                                        ▶
            S11_Q => {
                if eng.frames_empty() {
                    Flow::Accept
                } else {
                    let call_site = eng.exit_frame();
                    Flow::Jump(call_return(call_site, rt::PortId::from_raw(0)))
                }
            }
            //   12   !   (program)                        14
            S12_Q => 'state: {
                if !cand_s12_q(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S14_Q)
            }
            //   14  └‣─  (expression_statement) [RecordSet(expr)]  17
            S14_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                loop {
                    if cand_s14_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        break 'state Flow::Backtrack;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S14_Q,
                    ))?;
                }
                finish_s14_q(eng, m)?
            }
            //   15  ─‣┘² _                                11
            S15_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Up(2))?.is_none() {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S11_Q)
            }
            //   17  └‣─  (Expr)                           00 : 15
            S17_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Down)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_checkpoint(m, rt::Checkpoint::call_retry(
                    eng.checkpoint_state(),
                    S17_Q,
                    rt::CallResume {
                        target: S00_EXPR,
                        call_site: S17_Q,
                        field: None,
                        policy: rt::SkipPolicy::Any,
                    },
                ))?;
                eng.enter_frame(m, S17_Q)?;
                Flow::Jump(S00_EXPR)
            }
            _ => unreachable!("ip {ip} is not a generated state"),
        })
    }

    /// Unwind the checkpoint stack: successor checkpoints resume dispatch, Call and
    /// Match checkpoints advance their sibling search and re-enter. Loops, never
    /// recurses — a run of exhausted retries unwinds in one call.
    fn backtrack<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str) -> Result<Unwound, rt::LimitExceeded> {
        'unwind: loop {
            m.charge(1)?;
            let Some((cp, snapshot)) = eng.pop_checkpoint() else {
                return Ok(Unwound::NoMatch);
            };
            eng.restore_checkpoint_state(m, cp.state, snapshot)?;

            match cp.resume {
                rt::Resume::Successor => return Ok(Unwound::Resumed(cp.ip)),

                // Call retry: advance to the next candidate satisfying the field
                // constraint, then re-enter the callee. Exhausted siblings keep
                // unwinding to an earlier checkpoint.
                rt::Resume::Call(resume) => {
                    if !eng.cursor_mut().continue_search(m, resume.policy)? {
                        continue 'unwind;
                    }
                    if let Some(field_id) = resume.field {
                        loop {
                            if eng.cursor_mut().has_field(m, field_id)? {
                                break;
                            }
                            if !eng.cursor_mut().continue_search(m, resume.policy)? {
                                continue 'unwind;
                            }
                        }
                    }
                    eng.push_checkpoint(m, rt::Checkpoint::call_retry(
                        eng.checkpoint_state(),
                        cp.ip,
                        resume,
                    ))?;
                    eng.enter_frame(m, resume.call_site)?;
                    return Ok(Unwound::Resumed(resume.target));
                }

                // Match retry: advance past the accepted-but-failed candidate and
                // re-run that state's sibling search from there.
                rt::Resume::Match => match match_retry(eng, m, _source, cp.ip)? {
                    Some(Flow::Jump(next)) => return Ok(Unwound::Resumed(next)),
                    Some(Flow::Accept) => return Ok(Unwound::Accepted),
                    Some(Flow::Backtrack) => unreachable!("finish never backtracks"),
                    None => continue 'unwind,
                },
            }
        }
    }

    fn match_retry<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Option<Flow>, rt::LimitExceeded> {
        match ip {
            S14_Q => {
                if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                    return Ok(None);
                }
                loop {
                    if cand_s14_q(eng, m)? {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(m, rt::SkipPolicy::Any)? {
                        return Ok(None);
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(m, rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S14_Q,
                    ))?;
                }
                finish_s14_q(eng, m).map(Some)
            }
            _ => unreachable!("match-retry checkpoint ip {ip} must address a sibling-search Match"),
        }
    }

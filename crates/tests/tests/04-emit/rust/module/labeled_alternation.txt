use ::plotnik_runtime as rt;

const _: () = assert!(rt::RUNTIME_ABI == 0, "runtime interface mismatch");

use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'t> {
    pub expr: QExpr<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum QExpr<'t> {
    Id { id: Node<'t> },
    Num { num: Node<'t> },
}

impl<'t> Q<'t> {
    /// Apply `Q` once at a suitable node. A wrong node kind is no-match.
    pub fn exec(
        node: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        matcher::GRAMMAR.verify(node);
        let mut eng = rt::MatcherEngine::new(node.walk());
        let mut scratch = rt::DecoderScratch::default();
        Self::apply(&mut eng, source, &mut scratch)
    }

    /// Search inclusively in source-order DFS preorder, skipping descendants of matches.
    /// An error ends iteration.
    pub fn each<'s>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<Item = ::core::result::Result<Self, rt::LimitExceeded>>
    + use<'t, 's> {
        let mut eng = rt::MatcherEngine::new(scope.walk());
        let mut scratch = rt::DecoderScratch::default();
        rt::Traversal::new(scope, move |node: rt::Node<'t>| {
            matcher::GRAMMAR.verify(node);
            eng.reset(node);
            Self::apply(&mut eng, source, &mut scratch)
        })
    }

    /// Consume the first item of `each`.
    pub fn find(
        scope: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        <Self as rt::Execute<'_, '_>>::find(scope, source)
    }

    /// Collect `each`, returning an error rather than a partial vector.
    pub fn find_all(
        scope: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::std::vec::Vec<Self>, rt::LimitExceeded> {
        <Self as rt::Execute<'_, '_>>::find_all(scope, source)
    }

    fn apply(
        eng: &mut rt::MatcherEngine<'t>,
        source: &str,
        scratch: &mut rt::DecoderScratch,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        let Some(journal) = matcher::q_apply(eng, source)? else {
            return Ok(None);
        };

        let mut decoder = rt::ResultDecoder::new(journal.output_events(), source, scratch);
        let value = decode_q(&mut decoder);
        decoder.finish();
        Ok(Some(value))
    }
}

impl<'t, 's> rt::Execute<'t, 's> for Q<'t> {
    type Item = Self;

    fn exec(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        Self::exec(node, source)
    }

    fn each(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<Item = ::core::result::Result<Self, rt::LimitExceeded>>
    {
        Self::each(scope, source)
    }
}

/// Decode one committed `Q` value.
fn decode_q<'t, 's>(decoder: &mut rt::ResultDecoder<'_, 't, 's>) -> Q<'t> {
    decoder.expect_record_open();
    let v0 = decode_q_expr(decoder);
    match decoder.expect_record_set() {
        4 => {},
        other => unreachable!("journal shape proven at emit: `Q` has no member index {other}"),
    }
    decoder.expect_record_close();
    Q {
        expr: v0,
    }
}

/// Decode one committed `QExpr` value.
fn decode_q_expr<'t, 's>(decoder: &mut rt::ResultDecoder<'_, 't, 's>) -> QExpr<'t> {
    match decoder.expect_variant_open() {
        // Id
        2 => {
            let v0 = decoder.expect_node();
            match decoder.expect_record_set() {
                0 => {},
                other => {
                    unreachable!(
                        "journal shape proven at emit: `QExpr` has no member index {other}"
                    )
                }
            }
            decoder.expect_variant_close();
            QExpr::Id {
                id: v0,
            }
        }
        // Num
        3 => {
            let v0 = decoder.expect_node();
            match decoder.expect_record_set() {
                1 => {},
                other => {
                    unreachable!(
                        "journal shape proven at emit: `QExpr` has no member index {other}"
                    )
                }
            }
            decoder.expect_variant_close();
            QExpr::Num {
                num: v0,
            }
        }
        other => {
            unreachable!(
                "journal shape proven at emit: `QExpr` has no variant case at member index {other}"
            )
        }
    }
}

mod matcher {
    use ::plotnik_runtime as rt;
    use rt::Flow;

    pub(super) static GRAMMAR: rt::Grammar = rt::Grammar {
        kinds: &[
            (1, "identifier", true),
            (114, "number", true),
            (134, "program", true),
            (150, "expression_statement", true),
        ],
        fields: &[
        ],
        identity: None,
    };

    pub(super) fn q_apply<'a, 't>(eng: &'a mut rt::MatcherEngine<'t>, source: &str) -> Result<Option<&'a rt::MatchJournal<'t>>, rt::LimitExceeded> {
        rt::Matcher { runner: run_0 }.run(eng, source, 0, rt::EntryBoundary::new(true, false))
    }

    // Q

    pub(crate) fn run_0(eng: &mut rt::MatcherEngine<'_>, source: &str, mut block: u32) -> Flow {
        let _ = (&eng, &source);
        'run: loop {
            return match block {
                0 => {
                    //   02   !   (program)                        04
                    let node = eng.node();
                    if node.kind_id() != 134 || !node.is_named() { return Flow::Backtrack; }
                    //   04  └‣─  (expression_statement)           12
                    if eng.cursor_mut().navigate(rt::Nav::Down).is_none() { return Flow::Backtrack; }
                    loop {
                        let node = eng.node();
                        if !(node.kind_id() != 150 || !node.is_named()) { break; }
                        if !eng.cursor_mut().continue_search(rt::SkipPolicy::Any) { return Flow::Backtrack; }
                    }
                    if rt::SkipPolicy::Any.admits(|| eng.node()) {
                        eng.push_checkpoint(rt::Checkpoint::new(eng.checkpoint_state(), 7u32));
                    }
                    block = 6;
                    continue 'run;
                }
                1 => {
                    //   05  ─‣┘² _                                01          [RecordSet(expr)]
                    if eng.cursor_mut().navigate(rt::Nav::Up(2)).is_none() { return Flow::Backtrack; }
                    eng.emit_output_event(|_| rt::JournalEvent::RecordSet(4));
                    //   01       return
                    Flow::Return(rt::PortId::from_raw(0))
                }
                2 => {
                    //   07   !   (identifier)                     05          [VariantOpen(Id) Node RecordSet(id) VariantClose]
                    let node = eng.node();
                    if node.kind_id() != 1 || !node.is_named() { return Flow::Backtrack; }
                    eng.emit_output_event(|_| rt::JournalEvent::VariantOpen(2));
                    eng.emit_output_event(|c| rt::JournalEvent::Node(c.node()));
                    eng.emit_output_event(|_| rt::JournalEvent::RecordSet(0));
                    eng.emit_output_event(|_| rt::JournalEvent::VariantClose);
                    block = 1;
                    continue 'run;
                }
                3 => {
                    //   08   !   (number)                         05          [VariantOpen(Num) Node RecordSet(num) VariantClose]
                    let node = eng.node();
                    if node.kind_id() != 114 || !node.is_named() { return Flow::Backtrack; }
                    eng.emit_output_event(|_| rt::JournalEvent::VariantOpen(3));
                    eng.emit_output_event(|c| rt::JournalEvent::Node(c.node()));
                    eng.emit_output_event(|_| rt::JournalEvent::RecordSet(1));
                    eng.emit_output_event(|_| rt::JournalEvent::VariantClose);
                    block = 1;
                    continue 'run;
                }
                4 => {
                    //   09   !   switch
                    //              (identifier)                   07
                    //              (number)                       08
                    let node = eng.node();
                    match node.kind_id() {
                        1 => {
                            if !node.is_named() { return Flow::Backtrack; }
                            eng.emit_output_event(|_| rt::JournalEvent::VariantOpen(2));
                            eng.emit_output_event(|c| rt::JournalEvent::Node(c.node()));
                            eng.emit_output_event(|_| rt::JournalEvent::RecordSet(0));
                            eng.emit_output_event(|_| rt::JournalEvent::VariantClose);
                            block = 1;
                            continue 'run;
                        }
                        114 => {
                            if !node.is_named() { return Flow::Backtrack; }
                            eng.emit_output_event(|_| rt::JournalEvent::VariantOpen(3));
                            eng.emit_output_event(|c| rt::JournalEvent::Node(c.node()));
                            eng.emit_output_event(|_| rt::JournalEvent::RecordSet(1));
                            eng.emit_output_event(|_| rt::JournalEvent::VariantClose);
                            block = 1;
                            continue 'run;
                        }
                        _ => Flow::Backtrack,
                    }
                }
                5 => {
                    //   11  ──!  _                                09, 11
                    if eng.cursor_mut().navigate(rt::Nav::NextExact).is_none() { return Flow::Backtrack; }
                    eng.push_successors(&[5u32]);
                    block = 4;
                    continue 'run;
                }
                6 => {
                    //   12  └─!  _                                09, 11
                    if eng.cursor_mut().navigate(rt::Nav::DownExact).is_none() { return Flow::Backtrack; }
                    eng.push_successors(&[5u32]);
                    block = 4;
                    continue 'run;
                }
                7 => {
                    if !eng.cursor_mut().continue_search(rt::SkipPolicy::Any) { return Flow::Backtrack; }
                    loop {
                        let node = eng.node();
                        if !(node.kind_id() != 150 || !node.is_named()) { break; }
                        if !eng.cursor_mut().continue_search(rt::SkipPolicy::Any) { return Flow::Backtrack; }
                    }
                    if rt::SkipPolicy::Any.admits(|| eng.node()) {
                        eng.push_checkpoint(rt::Checkpoint::new(eng.checkpoint_state(), 7u32));
                    }
                    block = 6;
                    continue 'run;
                }
                _ => unreachable!("invalid local runner block {block}"),
            };
        }
    }
}

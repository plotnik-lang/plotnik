// Generated Plotnik query module: typed result types, supplied-node execution
// points, per-type result decoders, and the compiled matcher (`mod matcher`).
// Matcher states mirror the NFA dump's labels 1:1 (`S{label}_{DEF}`), and every
// dispatch arm carries its instruction in the dump format.

use ::plotnik_runtime as rt;

pub const REQUIRED_RUNTIME_ABI: u32 = 0;
const _: () = assert!(
    rt::RUNTIME_ABI == REQUIRED_RUNTIME_ABI,
    "Plotnik runtime ABI mismatch"
);

#[derive(Clone, Copy)]
pub struct Q<'r, 's> {
    pub comments: &'r [&'s str],
    pub has_statement: bool,
}

const _: () = assert!(::core::mem::size_of::<Q<'static, 'static>>() <= rt::GENERATED_VALUE_CAP);

rt::result_family!(Q<'r, 's>);

impl rt::json::Json for Q<'_, '_> {
    fn shape(&self) -> rt::json::Shape<'_> {
        rt::json::Shape::Record(self)
    }
}

impl rt::json::Fields for Q<'_, '_> {
    fn count(&self) -> usize {
        2
    }

    fn field(&self, index: usize) -> (&'static str, &dyn rt::json::Json) {
        match index {
            0 => ("comments", &self.comments),
            1 => ("has_statement", &self.has_statement),
            _ => unreachable!("`Q` has 2 fields"),
        }
    }
}

impl<'r, 's> Q<'r, 's> {
    /// Apply `Q` once at a suitable node. A wrong node kind is no-match.
    pub fn exec<'t>(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<rt::Match<Self>>, rt::LimitExceeded> {
        rt::run::exec::<Self, matcher::QEntry>(node, source)
    }

    /// Search inclusively in source-order DFS preorder, skipping descendants of matches.
    /// Each item owns its result. Matching and traversal share compiled-in limits.
    /// An error ends iteration.
    pub fn each<'t>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<
        Item = ::core::result::Result<rt::Match<Self>, rt::LimitExceeded>,
    > + use<'r, 't, 's> {
        rt::run::each::<Self, matcher::QEntry>(scope, source)
    }

    /// Consume the first item of `each`.
    pub fn find<'t>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<rt::Match<Self>>, rt::LimitExceeded> {
        Self::each(scope, source).next().transpose()
    }

    /// Collect `each`, returning an error rather than a partial collection.
    pub fn find_all<'t>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<rt::Matches<Self>, rt::LimitExceeded> {
        rt::run::find_all::<Self, matcher::QEntry>(scope, source)
    }
}

impl<'r, 't, 's> rt::Execute<'t, 's> for Q<'r, 's> {
    type Item = rt::Match<Self>;
    type All = rt::Matches<Self>;

    fn exec(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<rt::Match<Self>>, rt::LimitExceeded> {
        Self::exec(node, source)
    }

    fn each(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<
        Item = ::core::result::Result<rt::Match<Self>, rt::LimitExceeded>,
    > {
        Self::each(scope, source)
    }

    fn find_all(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<rt::Matches<Self>, rt::LimitExceeded> {
        Self::find_all(scope, source)
    }
}

impl<'r, 't, 's> rt::Publish<'t, 's> for Q<'r, 's> {
    const SCHEMA: rt::Schema<'static> = publish::SCHEMA;
    const ROOT: u32 = 0;
    const LAYOUTS: &'static [::std::alloc::Layout] = publish::LAYOUTS;

    fn build<'a>(
        walk: &mut rt::Walk<'_, '_, 't, 's>,
        scope: &rt::Scope<'a>,
        meter: &mut rt::Meter<'_>,
    ) -> ::core::result::Result<Q<'a, 's>, rt::LimitExceeded>
    where
        Self: 'a,
    {
        let mut stacks = publish::Stacks0::new();
        stacks.run(walk, scope, meter)?;
        Ok(publish::pop(&mut stacks.t0))
    }

    fn json<'a>(value: &'a Q<'a, 's>) -> &'a dyn rt::json::Json
    where
        Self: 'a,
    {
        value
    }
}

/// Typed construction of result values from walk steps.
mod publish {
    use super::*;

    /// The logical result schema shared with the runtime walk and JSON.
    pub(super) const SCHEMA: rt::Schema<'static> = rt::Schema {
        types: &[
            rt::Ty::Record { fields: rt::Span::new(0, 2) },
            rt::Ty::Text,
            rt::Ty::List { element: 1, minimum: 0 },
            rt::Ty::Bool,
        ],
        fields: &[
            rt::Field { name: "comments", ty: 2 },
            rt::Field { name: "has_statement", ty: 3 },
        ],
        cases: &[],
        members: &[
            rt::Member::new(0, 0),
            rt::Member::new(0, 1),
        ],
    };

    /// Native layout of each schema type, for sizing result blocks.
    pub(super) const LAYOUTS: &[::std::alloc::Layout] = &[
        ::std::alloc::Layout::new::<Q<'static, 'static>>(),
        ::std::alloc::Layout::new::<&'static str>(),
        ::std::alloc::Layout::new::<&'static [&'static str]>(),
        ::std::alloc::Layout::new::<bool>(),
    ];

    /// One value stack per schema type, plus one writer stack per list type.
    pub(super) struct Stacks0<'a, 't, 's> {
        pub(super) t0: rt::Buffer<Q<'a, 's>>,
        pub(super) t1: rt::Buffer<&'s str>,
        pub(super) t2: rt::Buffer<&'a [&'s str]>,
        w2: rt::Buffer<rt::Slice<'a, &'s str>>,
        pub(super) t3: rt::Buffer<bool>,
        _lifetimes: ::core::marker::PhantomData<(&'a (), &'t (), &'s ())>,
    }

    impl<'a, 't, 's> Stacks0<'a, 't, 's> {
        pub(super) fn new() -> Self {
            Self {
                t0: rt::Buffer::new(),
                t1: rt::Buffer::new(),
                t2: rt::Buffer::new(),
                w2: rt::Buffer::new(),
                t3: rt::Buffer::new(),
                _lifetimes: ::core::marker::PhantomData,
            }
        }

        /// Apply walk steps until the walk ends.
        pub(super) fn run(
            &mut self,
            walk: &mut rt::Walk<'_, '_, 't, 's>,
            _scope: &rt::Scope<'a>,
            meter: &mut rt::Meter<'_>,
        ) -> Result<(), rt::LimitExceeded> {
            while let Some(step) = walk.next(meter)? {
                match step {
                    rt::Step::Close { ty: 0, .. } => self.close_0(meter)?,
                    rt::Step::Text { ty: 1, text } => self.t1.push(meter, text)?,
                    rt::Step::Open { ty: 2, len, .. } => self.w2.push(meter, _scope.slice(len))?,
                    rt::Step::Element { list: 2 } => {
                        let value = pop(&mut self.t1);
                        let writer = self.w2.last_mut().expect("an element belongs to an open list");
                        writer.write(meter, value)?;
                    }
                    rt::Step::Close { ty: 2, .. } => self.close_2(meter)?,
                    rt::Step::Bool { ty: 3, value } => self.t3.push(meter, value)?,
                    rt::Step::Open { .. } | rt::Step::Field { .. } => {}
                    other => unreachable!("the schema admits no step {other:?}"),
                }
            }
            Ok(())
        }

        #[inline(never)]
        fn close_0(&mut self, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let has_statement = pop(&mut self.t3);
            let comments = pop(&mut self.t2);
            self.t0.push(meter, Q { comments, has_statement })
        }

        #[inline(never)]
        fn close_2(&mut self, meter: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            let writer = self.w2.pop().expect("a list closes its writer");
            self.t2.push(meter, writer.finish())
        }
    }

    pub(super) fn pop<T>(stack: &mut rt::Buffer<T>) -> T {
        stack.pop().expect("the walk produced every value before its parent")
    }
}

/// The compiled matcher: engine machinery shielded from the query's
/// type namespace.
mod matcher {
    use ::plotnik_runtime as rt;

    /// The limit policy compiled into the safe entrypoints, resolved against
    /// each input's node count. Chosen at generation time, never at the call
    /// site: the query is trusted, the input is not.
    const LIMITS: rt::RuntimeLimitSpec = rt::RuntimeLimitSpec {
        fuel_limit: rt::Limit::Auto,
        memory: rt::Limit::Auto,
    };

    /// Node-kind ids baked into the candidate checks: `(id, name, is_named)`
    /// as the generation-time grammar defines them.
    const EXPECTED_KINDS: &[(u16, &str, bool)] = &[
        (108, "comment", true),
        (134, "program", true),
        (150, "expression_statement", true),
    ];

    /// Field ids baked into the field checks: `(id, name)`.
    const EXPECTED_FIELDS: &[(u16, &str)] = &[];

    /// A parser built from any other grammar version could renumber the baked
    /// kind/field ids and silently mis-match, so mismatches panic: version skew
    /// between the generation-time grammar and the runtime parser is a build
    /// mistake, not a runtime condition to recover from.
    pub(super) fn verify_language(node: rt::Node<'_>) {
        let language = node.language();
        for &(id, name, named) in EXPECTED_KINDS {
            let found = language.node_kind_for_id(id);
            if found != Some(name) || language.node_kind_is_named(id) != named {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where node kind {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
        for &(id, name) in EXPECTED_FIELDS {
            let found = language.field_name_for_id(id);
            if found != Some(name) {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where field {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
    }

    // Dense runtime state ids, in NFA label order.
    // Q:
    const S01_Q: u16 = 0;
    const S02_Q: u16 = 1;
    const S03_Q: u16 = 2;
    const S05_Q: u16 = 3;
    const S07_Q: u16 = 4;
    const S08_Q: u16 = 5;
    const S09_Q: u16 = 6;
    const S11_Q: u16 = 7;
    const S17_Q: u16 = 8;
    const S19_Q: u16 = 9;
    const S20_Q: u16 = 10;
    const S21_Q: u16 = 11;
    const S23_Q: u16 = 12;
    const S25_Q: u16 = 13;
    const S26_Q: u16 = 14;
    const S31_Q: u16 = 15;

    pub(super) struct QEntry;

    impl rt::Entry for QEntry {
        const LIMITS: rt::RuntimeLimitSpec = LIMITS;

        fn verify_language(node: rt::Node<'_>) {
            verify_language(node);
        }

        fn run<'t>(
            eng: &mut rt::Engine<'t>,
            m: &mut rt::Meter<'_>,
            source: &str,
        ) -> Result<bool, rt::LimitExceeded> {
            run(eng, m, source, S02_Q, EntryBoundary::new(true, false))
        }
    }
    #[derive(Clone, Copy)]
    struct EntryBoundary {
        record: bool,
        node: bool,
    }

    impl EntryBoundary {
        const fn new(record: bool, node: bool) -> Self {
            Self { record, node }
        }

        fn begin(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordOpen)?;
            }
            Ok(())
        }

        fn finish(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.node {
                eng.emit(m, |cursor| rt::JournalEvent::Node(cursor.node()))?;
                return Ok(());
            }
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordClose)?;
            }
            Ok(())
        }
    }

    /// What a dispatched state hands back to the driver loop.
    enum Flow {
        /// Continue at this state.
        Jump(u16),
        /// The entrypoint accepted; the match journal is committed.
        Accept,
        /// The state failed; unwind the checkpoint stack.
        Backtrack,
    }

    /// How the backtrack unwind resumed execution.
    enum Unwound {
        Resumed(u16),
        Accepted,
        NoMatch,
    }

    fn run<'t>(
        eng: &mut rt::Engine<'t>,
        m: &mut rt::Meter<'_>,
        source: &str,
        entry: u16,
        boundary: EntryBoundary,
    ) -> Result<bool, rt::LimitExceeded> {
        boundary.begin(eng, m)?;
        let mut ip = entry;
        loop {
            m.charge(1)?;
            match dispatch(eng, m, source, ip)? {
                Flow::Jump(next) => ip = next,
                Flow::Accept => {
                    boundary.finish(eng, m)?;
                    eng.accepted_tape();
                    return Ok(true);
                }
                Flow::Backtrack => match backtrack(eng, m, source)? {
                    Unwound::Resumed(next) => ip = next,
                    Unwound::Accepted => {
                        boundary.finish(eng, m)?;
                        eng.accepted_tape();
                        return Ok(true);
                    }
                    Unwound::NoMatch => return Ok(false),
                },
            }
        }
    }

    fn dispatch<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Flow, rt::LimitExceeded> {
        Ok(match ip {
            //   01                                        ▶
            S01_Q => Flow::Accept,
            //   02   !   (program) [RecordSet(comments)]  31
            S02_Q => 'state: {
                if !cand_s02_q(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                eng.emit(m, |_| rt::JournalEvent::RecordSet(0))?; // RecordSet(comments)
                Flow::Jump(S31_Q)
            }
            //   03  ─‣┘  _                                01
            S03_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::Up(1))?.is_none() {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S01_Q)
            }
            //   05   !   (expression_statement) [BoolValue(1)]  03
            S05_Q => 'state: {
                if !cand_s05_q(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                eng.bool_value(m, true)?;
                Flow::Jump(S03_Q)
            }
            //   07  ──!  _                                05, 07
            S07_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::NextExact)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_successors(m, &[S07_Q])?;
                Flow::Jump(S05_Q)
            }
            //   08  └─!  _                                05, 07
            S08_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::DownExact)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_successors(m, &[S07_Q])?;
                Flow::Jump(S05_Q)
            }
            //   09  -ε-  [BoolValue(0)]                   01
            S09_Q => {
                eng.bool_value(m, false)?;
                Flow::Jump(S01_Q)
            }
            //   11  -ε-  [RecordSet(has_statement)]       08, 09
            S11_Q => {
                eng.emit(m, |_| rt::JournalEvent::RecordSet(1))?; // RecordSet(has_statement)
                eng.push_successors(m, &[S09_Q])?;
                Flow::Jump(S08_Q)
            }
            //   17  -ε-  [BoolValue(0)]                   03
            S17_Q => {
                eng.bool_value(m, false)?;
                Flow::Jump(S03_Q)
            }
            //   19  -ε-  [RecordSet(has_statement)]       07, 17
            S19_Q => {
                eng.emit(m, |_| rt::JournalEvent::RecordSet(1))?; // RecordSet(has_statement)
                eng.push_successors(m, &[S17_Q])?;
                Flow::Jump(S07_Q)
            }
            //   20  -ε-  [ListClose]                      19
            S20_Q => {
                eng.emit(m, |_| rt::JournalEvent::ListClose)?;
                Flow::Jump(S19_Q)
            }
            //   21  -ε-  [ListClose]                      11
            S21_Q => {
                eng.emit(m, |_| rt::JournalEvent::ListClose)?;
                Flow::Jump(S11_Q)
            }
            //   23   !   (comment) [NodeText ArrayPush]   25, 20
            S23_Q => 'state: {
                if !cand_s23_q(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                eng.node_text(m)?;
                eng.emit(m, |_| rt::JournalEvent::ArrayPush)?;
                eng.push_successors(m, &[S20_Q])?;
                Flow::Jump(S25_Q)
            }
            //   25  ──!  _                                23, 25
            S25_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::NextExact)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_successors(m, &[S25_Q])?;
                Flow::Jump(S23_Q)
            }
            //   26  └─!  _                                23, 25
            S26_Q => 'state: {
                if eng.cursor_mut().navigate(m, rt::Nav::DownExact)?.is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_successors(m, &[S25_Q])?;
                Flow::Jump(S23_Q)
            }
            //   31  -ε-  [ListOpen]                       26, 21
            S31_Q => {
                eng.emit(m, |_| rt::JournalEvent::ListOpen)?;
                eng.push_successors(m, &[S21_Q])?;
                Flow::Jump(S26_Q)
            }
            _ => unreachable!("ip {ip} is not a generated state"),
        })
    }

    /// `S02_Q` candidate: `(program)`.
    #[inline]
    fn cand_s02_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (program)
        if node.kind_id() != 134 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S05_Q` candidate: `(expression_statement)`.
    #[inline]
    fn cand_s05_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (expression_statement)
        if node.kind_id() != 150 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// `S23_Q` candidate: `(comment)`.
    #[inline]
    fn cand_s23_q(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (comment)
        if node.kind_id() != 108 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// Unwind the checkpoint stack: successor checkpoints resume dispatch, Call and
    /// Match checkpoints advance their sibling search and re-enter. Loops, never
    /// recurses — a run of exhausted retries unwinds in one call.
    fn backtrack<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str) -> Result<Unwound, rt::LimitExceeded> {
        'unwind: loop {
            m.charge(1)?;
            let Some((cp, snapshot)) = eng.pop_checkpoint() else {
                return Ok(Unwound::NoMatch);
            };
            eng.restore_checkpoint_state(m, cp.state, snapshot)?;

            match cp.resume {
                rt::Resume::Successor => return Ok(Unwound::Resumed(cp.ip)),

                rt::Resume::Call(_) => unreachable!("matcher has no call checkpoints"),

                // Match retry: advance past the accepted-but-failed candidate and
                // re-run that state's sibling search from there.
                rt::Resume::Match => match match_retry(eng, m, _source, cp.ip)? {
                    Some(Flow::Jump(next)) => return Ok(Unwound::Resumed(next)),
                    Some(Flow::Accept) => return Ok(Unwound::Accepted),
                    Some(Flow::Backtrack) => unreachable!("finish never backtracks"),
                    None => continue 'unwind,
                },
            }
        }
    }

    fn match_retry(_eng: &mut rt::Engine<'_>, _m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Option<Flow>, rt::LimitExceeded> {
        unreachable!("match-retry checkpoint ip {ip} must address a sibling-search Match")
    }
}

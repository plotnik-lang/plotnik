  (expression_statement)? @has_statement :: bool

use ::plotnik_runtime as rt;

const _: () = assert!(rt::RUNTIME_ABI == 0, "runtime interface mismatch");

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'s> {
    pub comments: ::std::vec::Vec<&'s str>,
    pub has_statement: bool,
}

impl<'s> Q<'s> {
    /// Apply `Q` once at a suitable node. A wrong node kind is no-match.
    pub fn exec(
        node: rt::Node<'_>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        matcher::GRAMMAR.verify(node);
        let mut eng = rt::MatcherEngine::new(node.walk());
        let mut scratch = rt::DecoderScratch::default();
        Self::apply(&mut eng, source, &mut scratch)
    }

    /// Search inclusively in source-order DFS preorder, skipping descendants of matches.
    /// An error ends iteration.
    pub fn each<'t>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<Item = ::core::result::Result<Self, rt::LimitExceeded>>
    + use<'t, 's> {
        let mut eng = rt::MatcherEngine::new(scope.walk());
        let mut scratch = rt::DecoderScratch::default();
        rt::Traversal::new(scope, move |node: rt::Node<'t>| {
            matcher::GRAMMAR.verify(node);
            eng.reset(node);
            Self::apply(&mut eng, source, &mut scratch)
        })
    }

    /// Consume the first item of `each`.
    pub fn find(
        scope: rt::Node<'_>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        <Self as rt::Execute<'_, '_>>::find(scope, source)
    }

    /// Collect `each`, returning an error rather than a partial vector.
    pub fn find_all(
        scope: rt::Node<'_>,
        source: &'s str,
    ) -> ::core::result::Result<::std::vec::Vec<Self>, rt::LimitExceeded> {
        <Self as rt::Execute<'_, '_>>::find_all(scope, source)
    }

    fn apply(
        eng: &mut rt::MatcherEngine<'_>,
        source: &'s str,
        scratch: &mut rt::DecoderScratch,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        let Some(journal) = matcher::q_apply(eng, source)? else {
            return Ok(None);
        };

        let mut decoder = rt::ResultDecoder::new(journal.output_events(), source, scratch);
        let value = decode_q(&mut decoder);
        decoder.finish();
        Ok(Some(value))
    }
}

impl<'t, 's> rt::Execute<'t, 's> for Q<'s> {
    type Item = Self;

    fn exec(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        Self::exec(node, source)
    }

    fn each(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<Item = ::core::result::Result<Self, rt::LimitExceeded>>
    {
        Self::each(scope, source)
    }
}

/// Decode one committed `Q` value.
fn decode_q<'t, 's>(decoder: &mut rt::ResultDecoder<'_, 't, 's>) -> Q<'s> {
    decoder.expect_record_open();
    let mut v0 = None; // comments
    let mut v1 = None; // has_statement
    while !decoder.at_record_close() {
        match decoder.peek_record_set() {
            // comments
            0 => {
                v0 = Some({
                    decoder.expect_list_open();
                    let mut items0 = ::std::vec::Vec::new();
                    while !decoder.at_list_close() {
                        items0.push(decoder.expect_str());
                        decoder.expect_array_push();
                    }
                    decoder.expect_list_close();
                    items0
                })
            }
            // has_statement
            1 => v1 = Some(decoder.expect_bool()),
            other => unreachable!("journal shape proven at emit: `Q` has no member index {other}"),
        }
        decoder.expect_record_set();
    }
    decoder.expect_record_close();
    Q {
        comments: v0.expect("field-stability: every accepting path sets `comments`"),
        has_statement: v1.expect("field-stability: every accepting path sets `has_statement`"),
    }
}

mod matcher {
    use ::plotnik_runtime as rt;
    use rt::Flow;

    pub(super) static GRAMMAR: rt::Grammar = rt::Grammar {
        kinds: &[
            (108, "comment", true),
            (134, "program", true),
            (150, "expression_statement", true),
        ],
        fields: &[
        ],
        identity: None,
    };

    pub(super) fn q_apply<'a, 't>(eng: &'a mut rt::MatcherEngine<'t>, source: &str) -> Result<Option<&'a rt::MatchJournal<'t>>, rt::LimitExceeded> {
        rt::Matcher { runner: run_0 }.run(eng, source, 1, rt::EntryBoundary::new(true, false))
    }

    // Q

    pub(crate) fn run_0(eng: &mut rt::MatcherEngine<'_>, source: &str, mut block: u32) -> Flow {
        let _ = (&eng, &source);
        'run: loop {
            return match block {
                0 => {
                    //   01       return
                    Flow::Return(rt::PortId::from_raw(0))
                }
                1 => {
                    //   02   !   (program)                        29
                    let node = eng.node();
                    if node.kind_id() != 134 || !node.is_named() { return Flow::Backtrack; }
                    //   29  -ε-                                   24, 19      [ListOpen]
                    eng.emit_output_event(|_| rt::JournalEvent::ListOpen);
                    eng.push_successors(&[9u32]);
                    block = 12;

                2 => {
                    //   03  ─‣┘  _                                01

                    block = 0;

                3 => {

                    let node = eng.node();
                    if node.kind_id() != 150 || !node.is_named() { return Flow::Backtrack; }

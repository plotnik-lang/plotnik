// Generated Plotnik query module: typed result types, supplied-node execution
// points, per-type result decoders, and the compiled matcher (`mod matcher`).
// Matcher states mirror the NFA dump's labels 1:1 (`S{label}_{DEF}`), and every
// dispatch arm carries its instruction in the dump format.

use ::plotnik_runtime as rt;

pub const REQUIRED_RUNTIME_ABI: u32 = 0;
const _: () = assert!(
    rt::RUNTIME_ABI == REQUIRED_RUNTIME_ABI,
    "Plotnik runtime ABI mismatch"
);

#[derive(Clone, Copy)]
pub struct Probe;



impl Probe {
    /// Apply `Probe` once at a suitable node. A wrong node kind is no-match.
    pub fn exec<'t>(
        node: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::core::option::Option<rt::Node<'t>>, rt::LimitExceeded> {
        rt::exec_node::<matcher::ProbeEntry>(node, source)
    }

    /// Search inclusively in source-order DFS preorder, skipping descendants of matches.
    /// Matching and traversal share compiled-in limits. An error ends iteration.
    pub fn each<'t, 's>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<
        Item = ::core::result::Result<rt::Node<'t>, rt::LimitExceeded>,
    > + use<'t, 's> {
        rt::each_node::<matcher::ProbeEntry>(scope, source)
    }

    /// Consume the first item of `each`.
    pub fn find<'t>(
        scope: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::core::option::Option<rt::Node<'t>>, rt::LimitExceeded> {
        Self::each(scope, source).next().transpose()
    }

    /// Collect `each`, returning an error rather than a partial vector.
    pub fn find_all<'t>(
        scope: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<rt::Buffer<rt::Node<'t>>, rt::LimitExceeded> {
        rt::find_all_nodes::<matcher::ProbeEntry>(scope, source)
    }
}

impl<'t, 's> rt::Execute<'t, 's> for Probe {
    type Item = rt::Node<'t>;
    type All = rt::Buffer<rt::Node<'t>>;

    fn exec(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<rt::Node<'t>>, rt::LimitExceeded> {
        Self::exec(node, source)
    }

    fn each(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<
        Item = ::core::result::Result<rt::Node<'t>, rt::LimitExceeded>,
    > {
        Self::each(scope, source)
    }

    fn find_all(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<rt::Buffer<rt::Node<'t>>, rt::LimitExceeded> {
        Self::find_all(scope, source)
    }
}

/// The compiled matcher: engine machinery shielded from the query's
/// type namespace.
mod matcher {
    use ::plotnik_runtime as rt;

    /// The limit policy compiled into the safe entrypoints, resolved against
    /// each input's node count. Chosen at generation time, never at the call
    /// site: the query is trusted, the input is not.
    const LIMITS: rt::RuntimeLimitSpec = rt::RuntimeLimitSpec {
        fuel_limit: rt::Limit::Auto,
        memory: rt::Limit::Auto,
    };

    /// Node-kind ids baked into the candidate checks: `(id, name, is_named)`
    /// as the generation-time grammar defines them.
    const EXPECTED_KINDS: &[(u16, &str, bool)] = &[(134, "program", true)];

    /// Field ids baked into the field checks: `(id, name)`.
    const EXPECTED_FIELDS: &[(u16, &str)] = &[];

    /// A parser built from any other grammar version could renumber the baked
    /// kind/field ids and silently mis-match, so mismatches panic: version skew
    /// between the generation-time grammar and the runtime parser is a build
    /// mistake, not a runtime condition to recover from.
    pub(super) fn verify_language(node: rt::Node<'_>) {
        let language = node.language();
        for &(id, name, named) in EXPECTED_KINDS {
            let found = language.node_kind_for_id(id);
            if found != Some(name) || language.node_kind_is_named(id) != named {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where node kind {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
        for &(id, name) in EXPECTED_FIELDS {
            let found = language.field_name_for_id(id);
            if found != Some(name) {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where field {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
    }

    // Dense runtime state ids, in NFA label order.
    // Probe:
    const S1_PROBE: u16 = 0;
    const S2_PROBE: u16 = 1;

    pub(super) struct ProbeEntry;

    impl rt::Entry for ProbeEntry {
        const LIMITS: rt::RuntimeLimitSpec = LIMITS;

        fn verify_language(node: rt::Node<'_>) {
            verify_language(node);
        }

        fn run<'t>(
            eng: &mut rt::Engine<'t>,
            m: &mut rt::Meter<'_>,
            source: &str,
        ) -> Result<bool, rt::LimitExceeded> {
            run(eng, m, source, S2_PROBE, EntryBoundary::new(false, false))
        }
    }
    #[derive(Clone, Copy)]
    struct EntryBoundary {
        record: bool,
        node: bool,
    }

    impl EntryBoundary {
        const fn new(record: bool, node: bool) -> Self {
            Self { record, node }
        }

        fn begin(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordOpen)?;
            }
            Ok(())
        }

        fn finish(self, eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<(), rt::LimitExceeded> {
            if self.node {
                eng.emit(m, |cursor| rt::JournalEvent::Node(cursor.node()))?;
                return Ok(());
            }
            if self.record {
                eng.emit(m, |_| rt::JournalEvent::RecordClose)?;
            }
            Ok(())
        }
    }

    /// What a dispatched state hands back to the driver loop.
    enum Flow {
        /// Continue at this state.
        Jump(u16),
        /// The entrypoint accepted; the match journal is committed.
        Accept,
        /// The state failed; unwind the checkpoint stack.
        Backtrack,
    }

    /// How the backtrack unwind resumed execution.
    enum Unwound {
        Resumed(u16),
        Accepted,
        NoMatch,
    }

    fn run<'t>(
        eng: &mut rt::Engine<'t>,
        m: &mut rt::Meter<'_>,
        source: &str,
        entry: u16,
        boundary: EntryBoundary,
    ) -> Result<bool, rt::LimitExceeded> {
        boundary.begin(eng, m)?;
        let mut ip = entry;
        loop {
            m.charge(1)?;
            match dispatch(eng, m, source, ip)? {
                Flow::Jump(next) => ip = next,
                Flow::Accept => {
                    boundary.finish(eng, m)?;
                    eng.accepted_tape();
                    return Ok(true);
                }
                Flow::Backtrack => match backtrack(eng, m, source)? {
                    Unwound::Resumed(next) => ip = next,
                    Unwound::Accepted => {
                        boundary.finish(eng, m)?;
                        eng.accepted_tape();
                        return Ok(true);
                    }
                    Unwound::NoMatch => return Ok(false),
                },
            }
        }
    }

    fn dispatch<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Flow, rt::LimitExceeded> {
        Ok(match ip {
            //   1                                         ▶
            S1_PROBE => Flow::Accept,
            //   2   !   (program)                         1
            S2_PROBE => 'state: {
                if !cand_s2_probe(eng, m)? {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S1_PROBE)
            }
            _ => unreachable!("ip {ip} is not a generated state"),
        })
    }

    /// `S2_PROBE` candidate: `(program)`.
    #[inline]
    fn cand_s2_probe(eng: &mut rt::Engine<'_>, m: &mut rt::Meter<'_>) -> Result<bool, rt::LimitExceeded> {
        m.charge(1)?;
        let node = eng.node();
        // (program)
        if node.kind_id() != 134 || !node.is_named() {
            return Ok(false);
        }
        Ok(true)
    }

    /// Unwind the checkpoint stack: successor checkpoints resume dispatch, Call and
    /// Match checkpoints advance their sibling search and re-enter. Loops, never
    /// recurses — a run of exhausted retries unwinds in one call.
    fn backtrack<'t>(eng: &mut rt::Engine<'t>, m: &mut rt::Meter<'_>, _source: &str) -> Result<Unwound, rt::LimitExceeded> {
        'unwind: loop {
            m.charge(1)?;
            let Some((cp, snapshot)) = eng.pop_checkpoint() else {
                return Ok(Unwound::NoMatch);
            };
            eng.restore_checkpoint_state(m, cp.state, snapshot)?;

            match cp.resume {
                rt::Resume::Successor => return Ok(Unwound::Resumed(cp.ip)),

                rt::Resume::Call(_) => unreachable!("matcher has no call checkpoints"),

                // Match retry: advance past the accepted-but-failed candidate and
                // re-run that state's sibling search from there.
                rt::Resume::Match => match match_retry(eng, m, _source, cp.ip)? {
                    Some(Flow::Jump(next)) => return Ok(Unwound::Resumed(next)),
                    Some(Flow::Accept) => return Ok(Unwound::Accepted),
                    Some(Flow::Backtrack) => unreachable!("finish never backtracks"),
                    None => continue 'unwind,
                },
            }
        }
    }

    fn match_retry(_eng: &mut rt::Engine<'_>, _m: &mut rt::Meter<'_>, _source: &str, ip: u16) -> Result<Option<Flow>, rt::LimitExceeded> {
        unreachable!("match-retry checkpoint ip {ip} must address a sibling-search Match")
    }
}

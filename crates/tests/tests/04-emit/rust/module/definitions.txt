// Generated Plotnik query module: typed result types, supplied-node execution
// points, per-type result decoders, and the compiled matcher (`mod matcher`).
// Matcher states mirror the NFA dump's labels 1:1 (`S{label}_{DEF}`), and every
// dispatch arm carries its instruction in the dump format.

use ::plotnik_runtime as rt;

pub const REQUIRED_RUNTIME_ABI: u32 = 0;
const _: () = assert!(
    rt::RUNTIME_ABI == REQUIRED_RUNTIME_ABI,
    "Plotnik runtime ABI mismatch"
);

use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Inner<'t> {
    pub id: Node<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'t> {
    pub second: Inner<'t>,
}

impl<'t> Inner<'t> {
    /// Apply `Inner` once at a suitable node. A wrong node kind is no-match.
    pub fn exec(
        node: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        let mut budget = matcher::execution_budget(node);
        let mut eng = rt::Engine::new(node.walk());
        matcher::verify_language(node);
        let mut scratch = rt::DecoderScratch::default();
        Self::apply(&mut eng, source, &mut budget, &mut scratch)
    }

    /// Search inclusively in source-order DFS preorder, retaining nested matches.
    /// Matching and traversal share compiled-in limits. An error ends iteration.
    pub fn each<'s>(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<Item = ::core::result::Result<Self, rt::LimitExceeded>>
    + use<'t, 's> {
        let mut eng = rt::Engine::new(scope.walk());
        let mut verified = false;
        let mut scratch = rt::DecoderScratch::default();
        rt::Traversal::new(
            scope,
            matcher::execution_budget(scope),
            move |node: rt::Node<'t>, budget: &mut rt::ExecutionBudget| {
                if !verified {
                    matcher::verify_language(node);
                    verified = true;
                }

                eng.reset(node);
                Self::apply(&mut eng, source, budget, &mut scratch)
            },
        )
    }

    /// Consume the first item of `each`.
    pub fn find(
        scope: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        <Self as rt::Execute<'_, '_>>::find(scope, source)
    }

    /// Collect `each`, returning an error rather than a partial vector.
    pub fn find_all(
        scope: rt::Node<'t>,
        source: &str,
    ) -> ::core::result::Result<::std::vec::Vec<Self>, rt::LimitExceeded> {
        <Self as rt::Execute<'_, '_>>::find_all(scope, source)
    }

    fn apply(
        eng: &mut rt::Engine<'t>,
        source: &str,
        budget: &mut rt::ExecutionBudget,
        scratch: &mut rt::DecoderScratch,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        let Some(journal) = matcher::inner_apply(eng, source, budget)? else {
            return Ok(None);
        };

        let mut decoder = rt::ResultDecoder::new(journal.output_events(), source, scratch);
        let value = decode_inner(&mut decoder);
        decoder.finish();
        Ok(Some(value))
    }
}

impl<'t, 's> rt::Execute<'t, 's> for Inner<'t> {
    type Item = Self;

    fn exec(
        node: rt::Node<'t>,
        source: &'s str,
    ) -> ::core::result::Result<::core::option::Option<Self>, rt::LimitExceeded> {
        Self::exec(node, source)
    }

    fn each(
        scope: rt::Node<'t>,
        source: &'s str,
    ) -> impl ::core::iter::FusedIterator<Item = ::core::result::Result<Self, rt::LimitExceeded>>
    {
        Self::each(scope, source)
    }
}

/// Decode one committed `Inner` value.
fn decode_inner<'t, 's>(decoder: &mut rt::ResultDecoder<'_, 't, 's>) -> Inner<'t> {
    decoder.expect_record_open();
    let mut v0 = None; // id
    while !decoder.at_record_close() {
        match decoder.peek_record_set() {
            // id
            0 => v0 = Some(decoder.expect_node()),
            other => {
                unreachable!("journal shape proven at emit: `Inner` has no member index {other}")
            }
        }
        decoder.expect_record_set();
    }
    decoder.expect_record_close();
    Inner {
        id: v0.expect("field-stability: every accepting path sets `id`"),
    }
}

/// Decode one committed `Q` value.
fn decode_q<'t, 's>(decoder: &mut rt::ResultDecoder<'_, 't, 's>) -> Q<'t> {
    decoder.expect_record_open();
    let mut v0 = None; // second
    while !decoder.at_record_close() {
        match decoder.peek_record_set() {
            // second
            1 => v0 = Some(decode_inner(decoder)),
            other => unreachable!("journal shape proven at emit: `Q` has no member index {other}"),
        }
        decoder.expect_record_set();
    }
    decoder.expect_record_close();
    Q {
        second: v0.expect("field-stability: every accepting path sets `second`"),
    }
}

/// The compiled matcher: engine machinery shielded from the query's
/// type namespace.
mod matcher {
    use ::plotnik_runtime as rt;

    /// The limit policy compiled into the safe entrypoints, resolved against
    /// each input's node count. Chosen at generation time, never at the call
    /// site: the query is trusted, the input is not.
    const LIMITS: rt::RuntimeLimitSpec = rt::RuntimeLimitSpec {
        fuel_limit: rt::Limit::Auto,
        memory: rt::Limit::Auto,
    };

    /// No ceilings — what the unmetered journal entrypoints run under.
    const NO_LIMITS: rt::ResolvedRuntimeLimits = rt::ResolvedRuntimeLimits {
        fuel_limit: None,
        max_memory: None,
    };

    /// Resolve compiled-in limits once for a supplied application or search scope.
    pub(super) fn execution_budget(scope: rt::Node<'_>) -> rt::ExecutionBudget {
        let source_nodes = u32::try_from(scope.descendant_count()).unwrap_or(u32::MAX);
        rt::ExecutionBudget::new(LIMITS.resolve(source_nodes))
    }

    /// Node-kind ids baked into the candidate checks: `(id, name, is_named)`
    /// as the generation-time grammar defines them.
    const EXPECTED_KINDS: &[(u16, &str, bool)] = &[
        (1, "identifier", true),
        (134, "program", true),
        (150, "expression_statement", true),
    ];

    /// Field ids baked into the field checks: `(id, name)`.
    const EXPECTED_FIELDS: &[(u16, &str)] = &[];

    /// A parser built from any other grammar version could renumber the baked
    /// kind/field ids and silently mis-match, so mismatches panic: version skew
    /// between the generation-time grammar and the runtime parser is a build
    /// mistake, not a runtime condition to recover from.
    pub(super) fn verify_language(node: rt::Node<'_>) {
        let language = node.language();
        for &(id, name, named) in EXPECTED_KINDS {
            let found = language.node_kind_for_id(id);
            if found != Some(name) || language.node_kind_is_named(id) != named {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where node kind {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
        for &(id, name) in EXPECTED_FIELDS {
            let found = language.field_name_for_id(id);
            if found != Some(name) {
                panic!(
                    "grammar version skew: this query module was generated against a \
                     grammar where field {id} is {name:?}, but the tree's language \
                     says {found:?} — rebuild the module with the grammar of the parser \
                     that produced the tree",
                );
            }
        }
    }

    // Dense runtime state ids, in NFA label order.
    // Inner:
    const S01_INNER: u16 = 0;
    const S02_INNER: u16 = 1;
    const S03_INNER: u16 = 2;
    const S04_INNER: u16 = 3;
    // Q:
    const S06_Q: u16 = 4;
    const S07_Q: u16 = 5;
    const S08_Q: u16 = 6;
    const S10_Q: u16 = 7;
    const S11_Q: u16 = 8;
    const S13_Q: u16 = 9;
    const S14_Q: u16 = 10;

    /// Resolve a callee-local return port through the immutable call-site map.
    #[inline]
    fn call_return(call_site: u16, port: rt::PortId) -> u16 {
        match (call_site, port.to_byte()) {
            (S11_Q, 0) => S10_Q,
            (S14_Q, 0) => S13_Q,
            _ => unreachable!(
                "call site {call_site} has no return port {}",
                port.to_byte()
            ),
        }
    }

    /// Match the `Inner` entrypoint against `tree`. `Some` carries the committed
    /// match journal — the same event sequence the VM commits for this query.
    pub fn inner_journal<'t>(tree: &'t rt::Tree, source: &str) -> Option<rt::MatchJournal<'t>> {
        let entry = S02_INNER;
        let boundary = EntryBoundary::new(true, false);
        let outcome = run::<false, false, true>(tree, source, entry, boundary, NO_LIMITS);
        outcome.expect("an unmetered run cannot exceed a limit")
    }

    /// Apply one candidate using the operation's cumulative budget.
    pub(super) fn inner_apply<'a, 't>(
        eng: &'a mut rt::Engine<'t>,
        source: &str,
        budget: &mut rt::ExecutionBudget,
    ) -> Result<Option<&'a rt::MatchJournal<'t>>, rt::LimitExceeded> {
        run_at_node::<true, true>(
            eng,
            source,
            S02_INNER,
            EntryBoundary::new(true, false),
            budget,
        )
    }

    /// Match the `Q` entrypoint against `tree`. `Some` carries the committed
    /// match journal — the same event sequence the VM commits for this query.
    pub fn q_journal<'t>(tree: &'t rt::Tree, source: &str) -> Option<rt::MatchJournal<'t>> {
        let entry = S07_Q;
        let boundary = EntryBoundary::new(true, false);
        let outcome = run::<false, false, true>(tree, source, entry, boundary, NO_LIMITS);
        outcome.expect("an unmetered run cannot exceed a limit")
    }

    /// Apply one candidate using the operation's cumulative budget.
    pub(super) fn q_apply<'a, 't>(
        eng: &'a mut rt::Engine<'t>,
        source: &str,
        budget: &mut rt::ExecutionBudget,
    ) -> Result<Option<&'a rt::MatchJournal<'t>>, rt::LimitExceeded> {
        run_at_node::<true, true>(eng, source, S07_Q, EntryBoundary::new(true, false), budget)
    }
    #[derive(Clone, Copy)]
    struct EntryBoundary {
        record: bool,
        node: bool,
    }

    impl EntryBoundary {
        const fn new(record: bool, node: bool) -> Self {
            Self { record, node }
        }

        fn begin(self, eng: &mut rt::Engine<'_>) {
            if self.record {
                let _ = eng.emit_output_event(|_| rt::JournalEvent::RecordOpen);
            }
        }

        fn finish(self, eng: &mut rt::Engine<'_>) {
            if self.node {
                let _ = eng.emit_output_event(|cursor| rt::JournalEvent::Node(cursor.node()));
                return;
            }
            if self.record {
                let _ = eng.emit_output_event(|_| rt::JournalEvent::RecordClose);
            }
        }
    }

    /// What a dispatched state hands back to the driver loop.
    enum Flow {
        /// Continue at this state.
        Jump(u16),
        /// The entrypoint accepted; the match journal is committed.
        Accept,
        /// The state failed; unwind the checkpoint stack.
        Backtrack,
        /// Source-driven call state exceeded a fixed-width runtime capacity.
        CallFrameError(rt::CallFrameError),
    }

    /// How the backtrack unwind resumed execution.
    enum Unwound {
        Resumed(u16),
        Accepted,
        NoMatch,
        CallFrameError(rt::CallFrameError),
    }

    /// One dispatch loop serves every entrypoint; `entry` selects the definition
    /// body and `boundary` supplies its root result effects.
    /// The metering constants come from the compiled-in policy. A fully unbounded
    /// policy skips budget accounting. Memory checks read `heap_bytes` only when
    /// memory is bounded. Metered dispatches use the same budget as the VM.
    /// `RECORD_OUTPUT_EVENTS` disables output allocation for match-only entries. (No
    /// let-chains: generated code targets the embedding crate's edition.)
    fn run<
        't,
        const METERED_FUEL: bool,
        const METERED_MEMORY: bool,
        const RECORD_OUTPUT_EVENTS: bool,
    >(
        tree: &'t rt::Tree,
        source: &str,
        entry: u16,
        boundary: EntryBoundary,
        limits: rt::ResolvedRuntimeLimits,
    ) -> Result<Option<rt::MatchJournal<'t>>, rt::LimitExceeded> {
        let mut budget = rt::ExecutionBudget::new(limits);
        let mut eng = if RECORD_OUTPUT_EVENTS {
            rt::Engine::new(tree.root_node().walk())
        } else {
            rt::Engine::new_match_only(tree.root_node().walk())
        };
        verify_language(eng.node());
        let accepted = run_at_node::<METERED_FUEL, METERED_MEMORY>(
            &mut eng,
            source,
            entry,
            boundary,
            &mut budget,
        )?
        .is_some();
        Ok(accepted.then(|| eng.into_journal()))
    }

    fn run_at_node<'a, 't, const METERED_FUEL: bool, const METERED_MEMORY: bool>(
        eng: &'a mut rt::Engine<'t>,
        source: &str,
        entry: u16,
        boundary: EntryBoundary,
        budget: &mut rt::ExecutionBudget,
    ) -> Result<Option<&'a rt::MatchJournal<'t>>, rt::LimitExceeded> {
        boundary.begin(eng);
        let mut ip = entry;
        loop {
            if METERED_FUEL || METERED_MEMORY {
                // One matcher dispatch consumes one unit of the operation budget.
                // Memory-only policies still charge to retain the sample cadence.
                budget.charge()?;
                // The shared budget schedules live matcher-heap samples.
                // Unbounded memory policies skip sampling and heap measurement.
                if METERED_MEMORY && budget.memory_sample_due() {
                    let used = eng.heap_bytes();
                    match budget.limits().max_memory {
                        Some(max) if used > max => {
                            return Err(rt::LimitExceeded::Memory { used, limit: max });
                        }
                        _ => {}
                    }
                }
            }
            match dispatch(eng, source, ip) {
                Flow::Jump(next) => ip = next,
                Flow::Accept => {
                    boundary.finish(eng);
                    return Ok(Some(eng.accepted_journal()));
                }
                Flow::CallFrameError(error) => return Err(error.into()),
                Flow::Backtrack => match backtrack(eng, source) {
                    Unwound::Resumed(next) => ip = next,
                    Unwound::Accepted => {
                        boundary.finish(eng);
                        return Ok(Some(eng.accepted_journal()));
                    }
                    Unwound::NoMatch => return Ok(None),
                    Unwound::CallFrameError(error) => return Err(error.into()),
                },
            }
        }
    }

    fn dispatch<'t>(eng: &mut rt::Engine<'t>, _source: &str, ip: u16) -> Flow {
        match ip {
            //   01                                        ▶
            S01_INNER => {
                if eng.frames_empty() {
                    Flow::Accept
                } else {
                    let call_site = eng.exit_frame();
                    Flow::Jump(call_return(call_site, rt::PortId::from_raw(0)))
                }
            }
            //   02   !   (expression_statement)           04
            S02_INNER => 'state: {
                if !cand_s02_inner(eng) {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S04_INNER)
            }
            //   03  ─‣┘  _                                01
            S03_INNER => 'state: {
                if eng.cursor_mut().navigate(rt::Nav::Up(1)).is_none() {
                    break 'state Flow::Backtrack;
                }
                Flow::Jump(S01_INNER)
            }
            //   04  └‣─  (identifier) [Node RecordSet(id)]  03
            S04_INNER => 'state: {
                if eng.cursor_mut().navigate(rt::Nav::Down).is_none() {
                    break 'state Flow::Backtrack;
                }
                loop {
                    if cand_s04_inner(eng) {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(rt::SkipPolicy::Any) {
                        break 'state Flow::Backtrack;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S04_INNER,
                    ));
                }
                finish_s04_inner(eng)
            }
            //   06                                        ▶
            S06_Q => {
                if eng.frames_empty() {
                    Flow::Accept
                } else {
                    let call_site = eng.exit_frame();
                    Flow::Jump(call_return(call_site, rt::PortId::from_raw(0)))
                }
            }
            //   07   !   (program) [SuppressBegin]        14
            S07_Q => 'state: {
                if !cand_s07_q(eng) {
                    break 'state Flow::Backtrack;
                }
                eng.suppress_begin();
                Flow::Jump(S14_Q)
            }
            //   08  ─‣┘  _ [RecordSet(second)]            06
            S08_Q => 'state: {
                if eng.cursor_mut().navigate(rt::Nav::Up(1)).is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.emit_output_event(|_| rt::JournalEvent::RecordSet(1)); // RecordSet(second)
                Flow::Jump(S06_Q)
            }
            //   10  -ε-  [RecordClose]                    08
            S10_Q => {
                eng.emit_output_event(|_| rt::JournalEvent::RecordClose);
                Flow::Jump(S08_Q)
            }
            //   11  ─‣─  (Inner)                          02 : 10
            S11_Q => 'state: {
                if eng.cursor_mut().navigate(rt::Nav::Next).is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_checkpoint(rt::Checkpoint::call_retry(
                    eng.checkpoint_state(),
                    S11_Q,
                    rt::CallResume {
                        target: S02_INNER,
                        call_site: S11_Q,
                        field: None,
                        policy: rt::SkipPolicy::Any,
                    },
                ));
                if let Err(error) = eng.enter_frame(S11_Q) {
                    return Flow::CallFrameError(error);
                }
                Flow::Jump(S02_INNER)
            }
            //   13  -ε-  [SuppressEnd RecordOpen]         11
            S13_Q => {
                eng.suppress_end();
                eng.emit_output_event(|_| rt::JournalEvent::RecordOpen);
                Flow::Jump(S11_Q)
            }
            //   14  └‣─  (Inner)                          02 : 13
            S14_Q => 'state: {
                if eng.cursor_mut().navigate(rt::Nav::Down).is_none() {
                    break 'state Flow::Backtrack;
                }
                eng.push_checkpoint(rt::Checkpoint::call_retry(
                    eng.checkpoint_state(),
                    S14_Q,
                    rt::CallResume {
                        target: S02_INNER,
                        call_site: S14_Q,
                        field: None,
                        policy: rt::SkipPolicy::Any,
                    },
                ));
                if let Err(error) = eng.enter_frame(S14_Q) {
                    return Flow::CallFrameError(error);
                }
                Flow::Jump(S02_INNER)
            }
            _ => unreachable!("ip {ip} is not a generated state"),
        }
    }

    /// `S02_INNER` candidate: `(expression_statement)`.
    #[inline]
    fn cand_s02_inner(eng: &rt::Engine<'_>) -> bool {
        let node = eng.node();
        // (expression_statement)
        if node.kind_id() != 150 || !node.is_named() {
            return false;
        }
        true
    }

    /// `S04_INNER` candidate: `(identifier)`.
    #[inline]
    fn cand_s04_inner(eng: &rt::Engine<'_>) -> bool {
        let node = eng.node();
        // (identifier)
        if node.kind_id() != 1 || !node.is_named() {
            return false;
        }
        true
    }

    /// `S07_Q` candidate: `(program)`.
    #[inline]
    fn cand_s07_q(eng: &rt::Engine<'_>) -> bool {
        let node = eng.node();
        // (program)
        if node.kind_id() != 134 || !node.is_named() {
            return false;
        }
        true
    }

    /// `S04_INNER` post-acceptance: effects, then successor dispatch. Shared by the dispatch
    /// path and the match-retry resume, so a retried candidate emits exactly
    /// what the original acceptance would have emitted.
    #[inline]
    fn finish_s04_inner(eng: &mut rt::Engine<'_>) -> Flow {
        eng.emit_output_event(|c| rt::JournalEvent::Node(c.node()));
        eng.emit_output_event(|_| rt::JournalEvent::RecordSet(0)); // RecordSet(id)
        Flow::Jump(S03_INNER)
    }

    /// Unwind the checkpoint stack: successor checkpoints resume dispatch, Call and
    /// Match checkpoints advance their sibling search and re-enter. Loops, never
    /// recurses — a run of exhausted retries unwinds in one call.
    fn backtrack<'t>(eng: &mut rt::Engine<'t>, _source: &str) -> Unwound {
        'unwind: loop {
            let Some((cp, snapshot)) = eng.pop_checkpoint() else {
                return Unwound::NoMatch;
            };
            eng.restore_checkpoint_state(cp.state, snapshot);

            match cp.resume {
                rt::Resume::Successor => return Unwound::Resumed(cp.ip),

                // Call retry: advance to the next candidate satisfying the field
                // constraint, then re-enter the callee. Exhausted siblings keep
                // unwinding to an earlier checkpoint.
                rt::Resume::Call(resume) => {
                    if !eng.cursor_mut().continue_search(resume.policy) {
                        continue 'unwind;
                    }
                    if let Some(field_id) = resume.field {
                        loop {
                            if eng.cursor().field_id() == Some(field_id) {
                                break;
                            }
                            if !eng.cursor_mut().continue_search(resume.policy) {
                                continue 'unwind;
                            }
                        }
                    }
                    eng.push_checkpoint(rt::Checkpoint::call_retry(
                        eng.checkpoint_state(),
                        cp.ip,
                        resume,
                    ));
                    if let Err(error) = eng.enter_frame(resume.call_site) {
                        return Unwound::CallFrameError(error);
                    }
                    return Unwound::Resumed(resume.target);
                }

                // Match retry: advance past the accepted-but-failed candidate and
                // re-run that state's sibling search from there.
                rt::Resume::Match => match match_retry(eng, _source, cp.ip) {
                    Some(Flow::Jump(next)) => return Unwound::Resumed(next),
                    Some(Flow::Accept) => return Unwound::Accepted,
                    Some(Flow::CallFrameError(error)) => {
                        return Unwound::CallFrameError(error);
                    }
                    Some(Flow::Backtrack) => unreachable!("finish never backtracks"),
                    None => continue 'unwind,
                },
            }
        }
    }

    fn match_retry<'t>(eng: &mut rt::Engine<'t>, _source: &str, ip: u16) -> Option<Flow> {
        match ip {
            S04_INNER => {
                if !eng.cursor_mut().continue_search(rt::SkipPolicy::Any) {
                    return None;
                }
                loop {
                    if cand_s04_inner(eng) {
                        break;
                    }
                    if !eng.cursor_mut().continue_search(rt::SkipPolicy::Any) {
                        return None;
                    }
                }
                if rt::SkipPolicy::Any.admits(&eng.node()) {
                    eng.push_checkpoint(rt::Checkpoint::match_retry(
                        eng.checkpoint_state(),
                        S04_INNER,
                    ));
                }
                Some(finish_s04_inner(eng))
            }
            _ => unreachable!("match-retry checkpoint ip {ip} must address a sibling-search Match"),
        }
    }
}

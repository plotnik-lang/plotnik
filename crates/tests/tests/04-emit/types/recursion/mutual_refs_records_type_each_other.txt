Arr = (array
  (Obj)? @obj
)

Obj = (object
  (pair
    value: (Arr)? @arr
  )
)

Q = (program
  (expression_statement (Arr) @top)
)
---------------------- RUST ----------------------
#[derive(Clone, Copy)]
pub struct Obj<'r> {
    pub arr: ::core::option::Option<&'r Arr<'r>>,
}

#[derive(Clone, Copy)]
pub struct Arr<'r> {
    pub obj: ::core::option::Option<&'r Obj<'r>>,
}

#[derive(Clone, Copy)]
pub struct Q<'r> {
    pub top: Arr<'r>,
}

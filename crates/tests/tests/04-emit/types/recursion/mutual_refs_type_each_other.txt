use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub enum B<'r, 't> {
    Leaf { n: Node<'t> },
    Wrap { a: &'r A<'r, 't> },
}

#[derive(Clone, Copy)]
pub enum A<'r, 't> {
    Base { id: Node<'t> },
    Nest { b: &'r B<'r, 't> },
}

#[derive(Clone, Copy)]
pub struct Q<'r, 't> {
    pub root: A<'r, 't>,
}

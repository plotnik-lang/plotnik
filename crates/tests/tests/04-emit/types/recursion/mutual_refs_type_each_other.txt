use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum B<'t> {
    Leaf { n: Node<'t> },
    Wrap { a: ::std::boxed::Box<A<'t>> },
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum A<'t> {
    Base { id: Node<'t> },
    Nest { b: ::std::boxed::Box<B<'t>> },
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'t> {
    pub root: A<'t>,
}

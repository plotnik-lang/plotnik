use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum Expr<'t> {
    Lit {
        value: Node<'t>,
    },
    Binary {
        left: ::std::boxed::Box<Expr<'t>>,
        right: ::std::boxed::Box<Expr<'t>>,
    },
}

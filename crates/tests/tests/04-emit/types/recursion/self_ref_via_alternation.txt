use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub enum Expr<'r, 't> {
    Lit {
        value: Node<'t>,
    },
    Binary {
        left: &'r Expr<'r, 't>,
        right: &'r Expr<'r, 't>,
    },
}

use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Inner<'t> {
    pub call: Node<'t>,
}

#[derive(Clone, Copy)]
pub struct Q<'t> {
    pub item: QItem<'t>,
}

#[derive(Clone, Copy)]
pub enum QItem<'t> {
    Call,
    Decl { d: Node<'t> },
}

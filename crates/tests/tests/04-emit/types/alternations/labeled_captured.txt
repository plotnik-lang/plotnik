Q = [
  Str: (string) @s
  Num: (number) @n
] @result
---------------------- RUST ----------------------
use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'t> {
    pub result: QResult<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum QResult<'t> {
    Str { s: Node<'t> },
    Num { n: Node<'t> },
}

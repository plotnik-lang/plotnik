Entry = [
  Id: (identifier) @id
  Num: (number) @num
]

Items = (Entry)*

Q = (program
  (expression_statement
    (array (Items) @items)
  )
)
---------------------- RUST ----------------------
use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub enum Entry<'t> {
    Id { id: Node<'t> },
    Num { num: Node<'t> },
}

#[derive(Clone, Copy)]
pub struct Q<'r, 't> {
    pub items: Items<'r, 't>,
}

pub type Items<'r, 't> = &'r [Entry<'t>];

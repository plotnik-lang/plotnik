  key: (_) @key
  value: (_) @val

  (expression_statement
    (parenthesized_expression
      (object (Rows) @rows)
    )
  )

use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Row<'t> {
    pub key: Node<'t>,
    pub val: Node<'t>,
}

#[derive(Clone, Copy)]
pub struct Q<'r, 't> {
    pub rows: Rows<'r, 't>,
}

pub type Rows<'r, 't> = &'r [Row<'t>];

use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Row<'t> {
    pub key: Node<'t>,
    pub val: Node<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'t> {
    pub rows: Rows<'t>,
}

pub type Rows<'t> = ::std::vec::Vec<Row<'t>>;

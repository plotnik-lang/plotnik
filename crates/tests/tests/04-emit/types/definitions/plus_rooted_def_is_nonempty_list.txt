use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Q<'r, 't> {
    pub ids: Ids<'r, 't>,
    pub runs: &'r [Ids<'r, 't>],
}

pub type Ids<'r, 't> = &'r [Node<'t>];

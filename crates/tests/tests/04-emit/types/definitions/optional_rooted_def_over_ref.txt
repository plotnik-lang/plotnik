use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Row<'t> {
    pub key: Node<'t>,
    pub val: Node<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Q<'t> {
    pub first: MaybeRow<'t>,
}

pub type MaybeRow<'t> = ::core::option::Option<Row<'t>>;

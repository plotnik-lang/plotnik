use ::plotnik_runtime::Node;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Name<'t> {
    pub name: Node<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Value<'t> {
    pub value: Node<'t>,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Both<'t> {
    pub key: Node<'t>,
    pub val: Node<'t>,
}

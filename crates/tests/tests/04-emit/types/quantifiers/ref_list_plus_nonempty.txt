use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Row<'t> {
    pub key: Node<'t>,
    pub val: Node<'t>,
}

#[derive(Clone, Copy)]
pub struct Q<'r, 't> {
    pub rows: &'r [Row<'t>],
}

Q = (program
  {
    (lexical_declaration) @decl
    (debugger_statement) @dbg
  }? @attrs
)
---------------------- RUST ----------------------
use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Q<'t> {
    pub attrs: ::core::option::Option<QAttrs<'t>>,
}

#[derive(Clone, Copy)]
pub struct QAttrs<'t> {
    pub decl: Node<'t>,
    pub dbg: Node<'t>,
}

Q = (program
  [
    Stmt: (expression_statement
      {
        (identifier) @name
      } @info
    )
    DECL: (lexical_declaration) @node
  ] @item
)
---------------------- RUST ----------------------
use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Q<'t> {
    pub item: QItem<'t>,
}

#[derive(Clone, Copy)]
pub enum QItem<'t> {
    Stmt { info: QItemStmtInfo<'t> },
    DECL { node: Node<'t> },
}

#[derive(Clone, Copy)]
pub struct QItemStmtInfo<'t> {
    pub name: Node<'t>,
}

Q = (program
  [
    Stmt: (expression_statement
      [
        Yes: (identifier)
        No: (number)
      ] @kind
    )
  ] @item
)
---------------------- RUST ----------------------
#[derive(Clone, Copy)]
pub struct Q {
    pub item: QItem,
}

#[derive(Clone, Copy)]
pub enum QItem {
    Stmt { kind: QItemStmtKind },
}

#[derive(Clone, Copy)]
pub enum QItemStmtKind {
    Yes,
    No,
}

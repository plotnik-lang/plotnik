Q = (program
  {
    (expression_statement) @stmt
  } @a :: Info
  {
    (expression_statement) @stmt
  } @b :: Info
)
---------------------- RUST ----------------------
use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Q<'t> {
    pub a: Info<'t>,
    pub b: Info<'t>,
}

#[derive(Clone, Copy)]
pub struct Info<'t> {
    pub stmt: Node<'t>,
}

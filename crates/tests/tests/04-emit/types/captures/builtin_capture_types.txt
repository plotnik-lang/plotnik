Q = (program
  (comment)? @doc :: text
  (comment)* @comments :: text
  (expression_statement)? @has_statement :: bool
)
---------------------- RUST ----------------------
#[derive(Clone, Copy)]
pub struct Q<'r, 's> {
    pub doc: ::core::option::Option<&'s str>,
    pub comments: &'r [&'s str],
    pub has_statement: bool,
}

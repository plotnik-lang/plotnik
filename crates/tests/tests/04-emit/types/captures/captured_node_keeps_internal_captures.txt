use ::plotnik_runtime::Node;

#[derive(Clone, Copy)]
pub struct Q<'t> {
    pub func: FunctionInfo<'t>,
    pub name: Node<'t>,
    pub body: Node<'t>,
}

pub type FunctionInfo<'t> = Node<'t>;
